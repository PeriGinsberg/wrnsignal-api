function ss(key: string): string {
    try {
        return sessionStorage.getItem(key) || ""
    } catch {
        return ""
    }
}

function getCookie(name: string): string {
    if (typeof document === "undefined") return ""
    const m = document.cookie.match(
        new RegExp("(?:^|; )" + name.replace(/[-.]/g, "\\$&") + "=([^;]*)")
    )
    return m ? decodeURIComponent(m[1]) : ""
}

function getAttributionSnapshot() {
    return {
        utm_source: ss("signal_utm_source"),
        utm_medium: ss("signal_utm_medium"),
        utm_campaign: ss("signal_utm_campaign"),
        utm_content: ss("signal_utm_content"),
        utm_term: ss("signal_utm_term"),
        landing_page: ss("signal_landing_page"),
        referrer: ss("signal_referrer"),
        fbclid: ss("signal_fbclid"),
        ttclid: ss("signal_ttclid"),
        gclid: ss("signal_gclid"),
        fbp: getCookie("_fbp"),
        fbc: getCookie("_fbc"),

    }
}

// Subset of attribution included as custom params on client-side Meta /
// gtag events. UTM-only — fbp/fbc/ttp + click IDs are already on the
// Stripe metadata path for CAPI server-side; no need to duplicate them
// onto the client-side event payload.
function getEventUtmParams() {
    return {
        utm_source: ss("signal_utm_source") || undefined,
        utm_medium: ss("signal_utm_medium") || undefined,
        utm_campaign: ss("signal_utm_campaign") || undefined,
        utm_content: ss("signal_utm_content") || undefined,
        utm_term: ss("signal_utm_term") || undefined,
    }
}

// Server-side mirror of the browser pixel. Fires /api/conversions/event
// with the same event_id used as fbq's eventID, so Meta dedups the pair
// inside its 7-day (event_name, event_id) window. Fire-and-forget — the
// browser pixel has already fired before this network call resolves.
type FunnelEventName = "Lead" | "ViewContent" | "AddToCart" | "InitiateCheckout"
type FunnelCustomData = {
    value?: number
    currency?: string
    verdict?: string
    score?: number | null
    cta_source?: string
    content_name?: string
    content_category?: string
}

function genEventId(): string {
    try {
        if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
            return crypto.randomUUID()
        }
    } catch {}
    return (
        Date.now().toString(36) +
        "-" +
        Math.random().toString(36).slice(2, 10)
    )
}

function sendServerEvent(args: {
    event_name: FunnelEventName
    event_id: string
    email?: string
    custom_data?: FunnelCustomData
}) {
    try {
        const sessionId =
            (typeof localStorage !== "undefined" &&
                localStorage.getItem("jobfit_session_id")) ||
            ""
        const attribution = getAttributionSnapshot()
        const body = {
            event_name: args.event_name,
            event_id: args.event_id,
            event_time_sec: Math.floor(Date.now() / 1000),
            email: args.email || undefined,
            session_id: sessionId,
            ...attribution,
            custom_data: args.custom_data || {},
        }
        const url = API_BASE + "/api/conversions/event"
        // fetch with keepalive:true gives sendBeacon-equivalent fire-and-
        // forget behavior (request survives page navigation) but defaults
        // to credentials:"omit" on cross-origin requests. sendBeacon would
        // be slightly simpler but forces credentials:"include", which then
        // requires the API to set Access-Control-Allow-Credentials:true —
        // a global change we don't want on a bearer-token-auth API.
        fetch(url, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
            keepalive: true,
            credentials: "omit",
        }).catch(() => {})
    } catch {}
}

function fileSizeLabel(bytes: number): string {
    if (bytes >= 1024 * 1024) {
        return (Math.round((bytes / 1024 / 1024) * 10) / 10) + " MB · ready"
    }
    return Math.round(bytes / 1024) + " KB · ready"
}

// Fires the analytics conversion event for "user submitted the free trial
// form". Uses sendBeacon so the event survives the page navigation. Mirrors
// the payload shape of free_trial_cta_click in landing_tracking_script.txt
// so prod analytics sees them as the same conversion regardless of which
// CTA path the user took. Best-effort: never throws, never blocks submit.
function fireFreeTrialCtaClick() {
    if (typeof window === "undefined") return
    try {
        const sessionId =
            (typeof crypto !== "undefined" && "randomUUID" in crypto
                ? localStorage.getItem("jobfit_session_id") || crypto.randomUUID()
                : null) || ""
        if (sessionId && !localStorage.getItem("jobfit_session_id")) {
            localStorage.setItem("jobfit_session_id", sessionId)
        }
        const payload = JSON.stringify({
            session_id: sessionId,
            page_name: "free_trial_cta_click",
            page_path: window.location.pathname,
            referrer: document.referrer || null,
            utm_source: ss("signal_utm_source") || null,
            utm_medium: ss("signal_utm_medium") || null,
            utm_campaign: ss("signal_utm_campaign") || null,
            cta_source: "landing_form_submit",
        })
        if (navigator && navigator.sendBeacon) {
            const blob = new Blob([payload], { type: "application/json" })
            navigator.sendBeacon(API_BASE + "/api/track", blob)
        } else {
            fetch(API_BASE + "/api/track", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: payload,
                keepalive: true,
            }).catch(() => {})
        }
    } catch {}
}

export default function SignalLandingPage(props: { style?: React.CSSProperties }) {
    const [resumeText, setResumeText] = React.useState("")
    const [resumeLabel, setResumeLabel] = React.useState("Upload your resume")
    const [resumeSub, setResumeSub] = React.useState("PDF (DOCX accepted)")
    const [resumeUploading, setResumeUploading] = React.useState(false)
    const [resumeError, setResumeError] = React.useState("")

    const [jdText, setJdText] = React.useState("")
    const [submitting, setSubmitting] = React.useState(false)
    const [submitError, setSubmitError] = React.useState("")

    const fileInputRef = React.useRef<HTMLInputElement | null>(null)

    React.useEffect(() => {
        const id = "v2-landing-styles"
        let el = document.getElementById(id) as HTMLStyleElement | null
        if (!el) {
            el = document.createElement("style")
            el.id = id
            document.head.appendChild(el)
        }
        el.textContent = STYLES
    }, [])

    // Resume carryover — hydrated when the user clicked Start Over on
    // the trial result page. Pre-fills resumeText state and updates the
    // upload box copy so the user knows their resume is already attached
    // and only the JD needs re-entry. Clicking the upload box still opens
    // the file picker for a different resume (handleFileChange takes over
    // and clears the carryover on successful upload).
    React.useEffect(() => {
        if (typeof window === "undefined") return
        try {
            const raw = sessionStorage.getItem(RESUME_CARRYOVER_KEY)
            if (!raw) return
            const parsed = JSON.parse(raw) as {
                resume_text?: string
                ts?: number
            }
            const ts = typeof parsed.ts === "number" ? parsed.ts : 0
            const fresh = ts && Date.now() - ts < RESUME_CARRYOVER_MAX_AGE_MS
            const text = String(parsed.resume_text || "").trim()
            if (!fresh || text.length < 100) {
                sessionStorage.removeItem(RESUME_CARRYOVER_KEY)
                return
            }
            setResumeText(text)
            setResumeLabel("Using your previous resume")
            setResumeSub("Click to upload a different file")
        } catch {}
    }, [])

    async function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
        const file = e.target.files?.[0]
        if (!file) return
        // Capture prior state so we can restore on upload failure — this
        // preserves the carryover resume when the user picks a new file
        // that fails to upload, rather than dumping them into an empty
        // upload-from-scratch state.
        const prevText = resumeText
        const prevLabel = resumeLabel
        const prevSub = resumeSub
        setResumeError("")
        setResumeUploading(true)
        setResumeLabel(file.name)
        setResumeSub("Reading file…")
        try {
            const fd = new FormData()
            fd.append("file", file)
            const res = await fetch(`${API_BASE}/api/resume-upload`, {
                method: "POST",
                body: fd,
            })
            const data = await res.json()
            if (!res.ok || !data?.text) {
                throw new Error(data?.error || "Upload failed. Try a different file.")
            }
            setResumeText(data.text)
            setResumeSub(fileSizeLabel(file.size))
            // New file uploaded successfully — drop any stale carryover
            // so a future Start Over writes the NEW resume, not the old.
            try {
                sessionStorage.removeItem(RESUME_CARRYOVER_KEY)
            } catch {}
        } catch (err: any) {
            // Restore prior state (empty OR carryover) so the user isn't
            // dumped to a blank upload after a failed re-upload attempt.
            setResumeText(prevText)
            setResumeLabel(prevText ? prevLabel : "Upload your resume")
            setResumeSub(prevText ? prevSub : "PDF (DOCX accepted)")
            setResumeError(err?.message || "Upload failed. Try a different file.")
        } finally {
            setResumeUploading(false)
            // Reset the input so re-selecting the same file still fires onChange.
            if (fileInputRef.current) fileInputRef.current.value = ""
        }
    }

    // Paste alternative to the file upload. Writes typed text to the same
    // resumeText state the upload path populates (last-writer-wins; no
    // mutual exclusion). When the user types real text, reset the upload
    // box label/sub so it stops claiming a previously-attached file is
    // still in play, and clear any stale upload error.
    function handleResumePaste(e: React.ChangeEvent<HTMLTextAreaElement>) {
        const value = e.target.value
        setResumeText(value)
        if (value) {
            setResumeLabel("Upload your resume")
            setResumeSub("PDF (DOCX accepted)")
        }
        setResumeError("")
    }

        e.preventDefault()
        setSubmitError("")
        if (resumeUploading) {
            setSubmitError("Hang on — your resume is still uploading.")
            return
        }
        if (!resumeText || resumeText.length < 100) {
            setSubmitError("Upload your resume first.")
            return
        }
        if (jdText.trim().length < 100) {
            setSubmitError("Paste the full job description — at least 100 characters.")
            return

        setSubmitting(true)
        try {

                resume_text: resumeText,
                job_description: jdText.trim(),

            }

        } catch {
            // sessionStorage write failure (private browsing, quota): fall

            // detect the missing handoff and route the user back here.
        }

        // Fire conversion pixel events. Meta Lead + GA4 free_scan_submitted.

            }
