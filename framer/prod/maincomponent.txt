/* The V5 prompt asks the model for a "3-5 WORD ALL-CAPS LABEL", so every
   keyword arrives shouting and has to be brought down here. Lowercasing
   wholesale would turn CRM into "Crm", so tokens on this list keep their case.
   It is a list, not a rule, and a keyword built on an acronym outside it will
   read wrong — the real fix is for the prompt to ask for sentence case in the
   first place, which is a server change and not this pass. */
// Keyed by the uppercase form, because that is what arrives; the value is the
// casing to restore. SaaS is why this is a map and not a set — the lookup key
// and the canonical spelling are not the same string.
const KEYWORD_ACRONYMS: Record<string, string> = {}
for (const a of [
    "AI", "API", "ATS", "B2B", "B2C", "CAD", "CPA", "CRM", "CX", "ERP", "ESG",
    "ETL", "FP&A", "GAAP", "GPA", "HR", "IT", "KPI", "M&A", "MBA", "ML", "NGO",
    "P&L", "PMP", "PR", "QA", "R&D", "ROI", "SaaS", "SEC", "SEM", "SEO", "SLA",
    "SQL", "UI", "UX", "VBA",
]) {
    KEYWORD_ACRONYMS[a.toUpperCase()] = a
}

function sentenceCase(raw: string): string {
    const s = (raw || "").trim()
    if (!s) return ""
    const words = s.split(/\s+/).map((w) => {
        const bare = w.replace(/[^A-Za-z&]/g, "")
        const canonical = KEYWORD_ACRONYMS[bare.toUpperCase()]
        if (canonical) return w.replace(bare, canonical)
        return w.toLowerCase()
    })
    const first = words[0]
    words[0] = first.charAt(0).toUpperCase() + first.slice(1)
    return words.join(" ")
}

/**
 * Split a rendered bullet into what we observed and what to do about it.
 *
 * The advice is NOT parsed out of the string when we can avoid it. V5 builds
 * the why string as `lead + " " + connection + " -> " + action`, and ships the
 * same three fields structurally on why_structured — so when the structured
 * object is present we read `.action` and never touch a substring.
 *
 * The string split is the fallback for pre-V5 cached rows, which have prose
 * but no structured array. It looks for the same ASCII " -> " the formatter
 * writes, and only splits on the first one.
 *
 * No arrow and no structured action means no advice. Nothing is invented: the
 * caller renders the observation alone.
 */
function splitAdvice(
    bullet: string,
    structured?: any
): { observation: string; advice: string } {
    const text = String(bullet || "").trim()

    const lead = String(structured?.lead ?? "").trim()
    const connection = String(structured?.connection ?? "").trim()
    const action = String(structured?.action ?? "").trim()
    if (action && (lead || connection)) {
        return {
            observation: [lead, connection].filter(Boolean).join(" "),
            advice: action,
        }
    }

    const at = text.indexOf(" -> ")
    if (at === -1) return { observation: text, advice: "" }
    return {
        observation: text.slice(0, at).trim(),
        advice: text.slice(at + 4).trim(),
    }
}

function safePreview(s: string, max = 1400) {
    const t = (s || "").trim()
    return t.length > max ? t.slice(0, max) + "\n…(truncated)" : t
}

function looksLikeHtml(text: string) {
    const t = (text || "").trim().toLowerCase()
    return (
        t.startsWith("<!doctype") ||
        t.startsWith("<html") ||
        t.includes("<head") ||
        t.includes("<body")
    )
}

async function postJsonWithDebug(
    url: string,
    payload: any,
    onDebug: (d: any) => void,
    accessToken?: string | null
) {
    const startedAt = Date.now()
    let res: Response

    const headers: Record<string, string> = {
        "Content-Type": "application/json",
    }
    if (accessToken) headers["Authorization"] = `Bearer ${accessToken}`

    try {
        res = await fetch(url, {
            method: "POST",
            headers,
            body: JSON.stringify(payload),
        })
    } catch (err: any) {
        onDebug({
            stage: "fetch_throw",
            url,
            ms: Date.now() - startedAt,
            errorName: err?.name,
            errorMessage: err?.message,
            errorStackPreview: safePreview(err?.stack || "", 900),
        })
        throw new Error(`Request failed. Please try again.`)
    }

    const ms = Date.now() - startedAt
    const contentType = res.headers.get("content-type") || ""
    const raw = await res.text()

    onDebug({
        stage: "fetch_response",
        url,
        ms,
        status: res.status,
        ok: res.ok,
        contentType,
        responsePreview: safePreview(raw),
    })

    if (looksLikeHtml(raw) || contentType.includes("text/html")) {
        throw new Error(`Unexpected response. Please try again.`)
    }

    let data: any = null
    try {
        data = JSON.parse(raw)
    } catch {
        throw new Error(`Unexpected response. Please try again.`)
    }

    if (!res.ok) {
        throw new Error(
            data?.detail || data?.error || `Request failed. Please try again.`
        )
    }

    return data
}

/* ---------- DIFF HIGHLIGHT (TEAL) ---------- */
function renderDiffTeal(before: string, after: string) {
    const normalize = (w: string) =>
        w
            .toLowerCase()
            .replace(/^[^a-z0-9]+|[^a-z0-9]+$/gi, "")
            .trim()

    const stop = new Set([
        "a",
        "an",
        "the",
        "and",
        "or",
        "but",
        "to",
        "of",
        "in",
        "on",
        "for",
        "with",
        "at",
        "by",
        "from",
        "as",
        "is",
        "are",
        "was",
        "were",
        "be",
        "been",
        "being",
        "this",
        "that",
        "these",
        "those",
        "it",
        "its",
        "you",
        "your",
        "we",
        "our",
        "i",
    ])

    const beforeWords = new Set(
        (before || "")
            .split(/\s+/)
            .map(normalize)
            .filter((w) => w && w.length >= 4 && !stop.has(w))
    )

    const parts = (after || "").split(/(\b[\w'/-]+\b)/g)

    return (
        <>
            {parts.map((p, i) => {
                if (!p || !/\b[\w'/-]+\b/.test(p))
                    return <span key={i}>{p}</span>

                const w = normalize(p)
                const shouldBold =
                    !!w && w.length >= 4 && !stop.has(w) && !beforeWords.has(w)

                // The diff mark is a pale-blue highlight now, not teal bold
                // text. Bolding a word inside a sentence to mean "this is new"
                // reads as emphasis; a highlight reads as a mark.
                return shouldBold ? (
                    <span key={i} style={diffMark}>
                        {p}
                    </span>
                ) : (
                    <span key={i}>{p}</span>
                )
            })}
        </>
    )
}

function extractJobContextFromJobText(jobText: string) {
    const text = String(jobText || "")
        .replace(/\r/g, "")
        .trim()
    if (!text) return { title: "", company: "" }

    const lines = text
        .split("\n")
        .map((l) => l.trim())
        .filter(Boolean)

    const firstN = lines.slice(0, 40)

    const clean = (s: string) =>
        String(s || "")
            .replace(/[–—]/g, "-")
            .replace(/\s+/g, " ")
            .trim()

    const looksLikeMarketingFluff = (s: string) => {
        const t = (s || "").toLowerCase()
        const phrases = [
            "best place to work",
            "great place to work",
            "fastest growing",
            "award",
            "top workplace",
            "fortune",
            "inc.",
            "recognition",
            "ranked",
            "named",
        ]
        if (phrases.some((p) => t.includes(p))) return true
        const quoteCount = (s.match(/["“”']/g) || []).length
        if (quoteCount >= 4) return true
        const commaCount = (s.match(/,/g) || []).length
        const words = s.split(/\s+/).filter(Boolean)
        const upperWords = words.filter((w) => /^[A-Z]{3,}$/.test(w)).length
        if (commaCount >= 2 && upperWords >= 3) return true
        return false
    }

    const looksLikeRealRoleTitle = (s: string) => {
        const t = (s || "").toLowerCase()
        const roleWords = [
            "intern",
            "analyst",
            "associate",
            "manager",
            "director",
            "coordinator",
            "specialist",
            "engineer",
            "consultant",
            "developer",
            "designer",
            "strategist",
            "assistant",
            "representative",
            "officer",
            "lead",
            "head",
        ]
        return roleWords.some((w) => t.includes(w))
    }

    const isJunkTitle = (s: string) => {
        const t = (s || "").toLowerCase()
        if (!t) return true
        if (t.length < 4) return true
        if (t.length > 90) return true
        if (looksLikeMarketingFluff(s)) return true
        if (
            t.includes("overview") ||
            t.includes("responsibilities") ||
            t.includes("qualifications") ||
            t.includes("requirements") ||
            t.includes("equal opportunity") ||
            t.includes("accommodation")
        )
            return true
        if (!looksLikeRealRoleTitle(s)) return true
        return false
    }

    const isJunkCompany = (s: string) => {
        const t = s.toLowerCase()
        if (!t) return true
        if (t.length < 2) return true
        if (t.length > 70) return true
        if (
            t.includes("overview") ||
            t.includes("position overview") ||
            t.includes("responsibilities") ||
            t.includes("qualifications") ||
            t.includes("equal opportunity")
        )
            return true
        return false
    }

    let company = ""

    for (const l of firstN.slice(0, 20)) {
        const m = l.match(/^([A-Z][A-Za-z0-9&.'-]{2,40})\s+Overview$/)
        if (m?.[1]) {
            const c = clean(m[1])
            if (!isJunkCompany(c)) company = company || c
        }
    }

    if (!company) {
        for (const l of firstN) {
            const m = l.match(
                /^([A-Z][A-Za-z0-9&.'-]{2,60})\s+is\s+(?:currently\s+)?looking\b/i
            )
            if (m?.[1]) {
                const c = clean(m[1])
                if (!isJunkCompany(c)) {
                    company = c
                    break
                }
            }
        }
    }

    if (!company) {
        for (const l of firstN) {
            const m = l.match(/^(Company|Employer)\s*:\s*(.+)$/i)
            if (m?.[2]) {
                const c = clean(m[2])
                if (!isJunkCompany(c)) {
                    company = c
                    break
                }
            }
        }
    }

    let title = ""

    for (const l of firstN) {
        const m = l.match(/^(Position Title|Job Title|Role)\s*:\s*(.+)$/i)
        if (m?.[2]) {
            const t = clean(m[2])
            if (!isJunkTitle(t)) {
                title = t
                break
            }
        }
    }

    if (!title) {
        for (const l of firstN) {
            const m = l.match(/\bas\s+(?:an?|the)\s+(.{4,80})$/i)
            if (m?.[1]) {
                const t = clean(m[1])
                if (!isJunkTitle(t)) {
                    title = t
                    break
                }
            }
        }
    }

    title = clean(title)
    company = clean(company)

    if (isJunkTitle(title)) title = ""
    if (isJunkCompany(company)) company = ""

    return { title, company }
}

/* ---------- TEXT HELPERS (SAFE) ---------- */
function pickString(obj: any, keys: string[]) {
    for (const k of keys) {
        const v = obj?.[k]
        if (typeof v === "string" && v.trim()) return v.trim()
    }
    return ""
}

function toSafeLine(x: any) {
    if (x === null || x === undefined) return ""
    if (typeof x === "string") return x.trim()
    if (typeof x === "number") return String(x)
    try {
        return JSON.stringify(x)
    } catch {
        return String(x)
    }
}

function pickFirstArray(...vals: any[]) {
    for (const v of vals) {
        if (Array.isArray(v) && v.length) return v
    }
    return []
}

/* ---------- COVER LETTER HELPERS ---------- */
function normalizeLetterBody(raw: string) {
    const s = String(raw || "")
        .replace(/\r/g, "")
        .trim()
    if (!s) return ""

    const lines = s.split("\n").map((l) => l.replace(/[ \t]+$/g, ""))
    const out: string[] = []
    let lastWasBlank = false

    for (const line of lines) {
        const isBlank = !line.trim()
        if (isBlank) {
            if (!lastWasBlank) out.push("")
            lastWasBlank = true
        } else {
            out.push(line.trimStart())
            lastWasBlank = false
        }
    }

    while (out.length && !out[0].trim()) out.shift()
    while (out.length && !out[out.length - 1].trim()) out.pop()

    return out.join("\n")
}

// Strip any leading salutation like "Dear X," so we never print two "Dear" lines.
function stripLeadingSalutation(raw: string) {
    const s = normalizeLetterBody(raw)
    if (!s) return ""

    const lines = s.split("\n")
    const firstNonBlankIndex = lines.findIndex((l) => l.trim().length > 0)
    if (firstNonBlankIndex === -1) return s

    const first = lines[firstNonBlankIndex].trim()
    const isDear = /^dear\s+.+/i.test(first)

    if (!isDear) return s

    // Remove that line and any immediate blank line after it
    const remaining = lines.slice(firstNonBlankIndex + 1)
    while (remaining.length && !remaining[0].trim()) remaining.shift()

    return normalizeLetterBody(remaining.join("\n"))
}

function stripExistingSignatureBlock(raw: string) {
    const s = normalizeLetterBody(raw)
    if (!s) return ""

    const signoffs = [
        "sincerely,",
        "sincerely",
        "best,",
        "best regards,",
        "best regards",
        "regards,",
        "regards",
        "thank you,",
        "thank you",
        "respectfully,",
        "respectfully",
    ]

    const lines = s.split("\n")
    const start = Math.max(0, lines.length - 10)

    for (let i = lines.length - 1; i >= start; i--) {
        const t = lines[i].trim().toLowerCase()
        if (signoffs.includes(t)) {
            return normalizeLetterBody(lines.slice(0, i).join("\n"))
        }
        for (const so of signoffs) {
            if (t.startsWith(so)) {
                return normalizeLetterBody(lines.slice(0, i).join("\n"))
            }
        }
    }

    return s
}

/* ---------- NETWORKING EXTRACTORS ---------- */
function extractNetworkingTarget(a: any) {
    if (!a) return ""

    const roles = a?.target_roles
    if (Array.isArray(roles) && roles.length) {
        const first =
            roles.find((x: any) => typeof x === "string" && x.trim()) ||
            roles.find(
                (x: any) => typeof x?.title === "string" && x.title.trim()
            )
        if (typeof first === "string") return first.trim()
        if (first?.title) return String(first.title).trim()
    } else if (typeof roles === "string" && roles.trim()) {
        return roles.trim()
    }

    const person = pickString(a, ["person_to_pick", "personToPick"])
    if (person) return person

    const terms = a?.search_terms
    if (Array.isArray(terms) && terms.length) {
        const firstTerm = terms.find(
            (x: any) => typeof x === "string" && x.trim()
        )
        if (firstTerm) return firstTerm.trim()
    } else if (typeof terms === "string" && terms.trim()) {
        return terms.trim()
    }

    return pickString(a, [
        "linkedin_title",
        "linkedinTitle",
        "search_title",
        "searchTitle",
        "target_title",
        "targetTitle",
        "role",
        "title",
        "persona",
        "label",
    ])
}

function extractNetworkingMessageParts(action: any) {
    const msg =
        action?.message ??
        action?.outreach_message ??
        action?.outreachMessage ??
        action?.copy ??
        action?.draft ??
        ""

    if (typeof msg === "string") {
        return { initial: msg.trim(), followUp: "" }
    }

    if (msg && typeof msg === "object") {
        const initial =
            pickString(msg, ["initial", "first", "message", "text", "body"]) ||
            pickString(msg?.content, ["text", "body", "content"]) ||
            pickString(msg?.message, ["text", "body", "content"]) ||
            ""

        const followUp =
            pickString(msg, ["follow_up", "followUp", "followup", "second"]) ||
            pickString(msg?.follow_up, ["text", "body", "content"]) ||
            pickString(msg?.followUp, ["text", "body", "content"]) ||
            ""

        return { initial, followUp }
    }

    return { initial: "", followUp: "" }
}

/**
 * THE PERSISTENT JOB CONTEXT BAR.
 *
 * One job, named once, with every screen that acts on it one click away. It is
 * the answer to "I ran JobFit, now where am I" — the tabs already existed, but
 * nothing on screen said which job they were about or that they were about the
 * same one.
 *
 * IDENTITY IS THE IDS, NOT THE TEXT. `runId` is what makes this a job rather
 * than a paste buffer. The bar does not render without one (or an application
 * id): if the server never wrote the row, there is nothing durable to be the
 * context OF, and offering navigation around an unsaved job would be a promise
 * the product cannot keep. Title and company are display only — they come from
 * the same jobCtx cascade the rest of the component reads, and they are never
 * what a destination is resolved by.
 *
 * NAVIGATION MUST NOT COMPUTE. Every tool destination goes through `onSelect`,
 * which is goTo / setTab and nothing else. It deliberately does NOT go through
 * proceedToPositioning or proceedToCoverLetter: those RUN the tool when no
 * result exists yet, and a context bar whose links can each cost thirty seconds
 * and an LLM call is not a context bar. Landing on an unrun tool shows that
 * tool's own landing state with its own Run button, which is the correct place
 * for that decision. Results already computed are held in the component's state
 * and survive every switch.
 *
 * JOB DETAILS IS THE ONE LINK THAT LEAVES. The tracker lives in the Next.js app,
 * not in this Framer component, so it is a full app switch carrying the session:
 * window.top with the token on the query string, because an <a href> would open
 * the dashboard inside the Framer iframe. It addresses the specific application,
 * never the list.
 *
 * NETWORKING IS ABSENT ON PURPOSE. It is out of scope for this task, and its
 * three surfaces do not agree on what a job even is yet.
 */
function JobContextBar({
    ctx,
    tab,
    runId,
    applicationId,
    onSelect,
    onJobDetails,
}: {
    ctx: { ctxTitle: string; ctxCompany: string }
    tab: TabKey
    runId: string | null
    applicationId: string | null
    onSelect: (t: TabKey) => void
    onJobDetails: () => void
}) {
    // No addressable job, no bar. Checked before anything else so the empty
    // case costs nothing and can never half-render.
    if (!runId && !applicationId) return null

    const lead = [ctx.ctxTitle, ctx.ctxCompany].filter(Boolean).join(" · ")

    const stops: { key: TabKey; label: string }[] = [
        { key: "jobfit", label: "JobFit" },
        { key: "positioning", label: "Positioning" },
        { key: "coverletter", label: "Cover Letter" },
    ]

    return (
        <div style={jcBar}>
            <div style={jcIdentity}>
                <div style={jcEyebrow}>Current job</div>
                {/* The fallback is not decoration. A cached result with thin
                    job_signals, or a run deep-linked before the title was
                    entered, can genuinely have neither field — and the bar
                    still has to name the row it is steering, so it says so
                    rather than rendering an empty line. */}
                <div style={jcLead}>{lead || "Untitled job"}</div>
            </div>

            <div style={jcStops}>
                {stops.map((s) => {
                    const active = tab === s.key
                    return (
                        <button
                            key={s.key}
                            type="button"
                            onClick={() => onSelect(s.key)}
                            aria-current={active ? "page" : undefined}
                            style={{
                                ...jcStop,
                                ...(active ? jcStopActive : null),
                            }}
                        >
                            {s.label}
                        </button>
                    )
                })}

                {/* Only when there is an application to open. Null is a normal
                    outcome — the scan's application write is best-effort, and
                    a deep-linked run cannot recover the id at all because
                    GET /api/runs/:id does not return it. A link that would
                    land on the tracker list instead of this job is worse than
                    no link, so it is absent rather than degraded. */}
                {applicationId ? (
                    <button
                        type="button"
                        onClick={onJobDetails}
                        style={{ ...jcStop, ...jcStopLeaves }}
                    >
                        Job Details ↗
                    </button>
                ) : null}
            </div>
        </div>
    )
}

                    if (accessToken && refreshToken) {

                            refresh_token: refreshToken,

    async function handleFetchUrl() {
        if (!jobUrl.trim()) return
        setUrlLoading(true)
        setUrlError("")
        setShowLinkedInHelper(false)
        setShowBlockedPopup(false)
        try {
            const res = await fetch(`${API_BASE}/api/parse-job-url`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ url: jobUrl.trim() }),
            })
            const data = await res.json()
            if (data.suggestion === "paste_text") {
                setShowBlockedPopup(true)
                setShowLinkedInHelper(true)
                return
            }
            if (data.error) {
                setUrlError(data.error)
                return
            }
            if (data.companyName) setManualCompanyName(data.companyName)
            if (data.jobTitle) setManualJobTitle(data.jobTitle)
            if (data.jobDescription) setJob(data.jobDescription)
            // Keep jobUrl — it gets sent to /api/jobfit as job_url
            // so the application dashboard auto-populates the link
            setUrlError("")
            setShowLinkedInHelper(false)
            setShowBlockedPopup(false)
        } catch {
            setUrlError("Something went wrong. Please paste the job description manually.")
        } finally {
            setUrlLoading(false)
        }
    }

    async function runJobFit() {
        clearErrorAndDebug()
        setLoadingJobFit(true)

        // Pre-capture context from job text (fallback) but do not overwrite existing
        setJobContext((prev) => {
            if (prev.title || prev.company) return prev
            const fromText = extractJobContextFromJobText(job)
            if (fromText.title || fromText.company) return fromText
            return prev
        })

        try {
            if (!hasJob) throw new Error("Paste a job description first.")
            // Job title and company name are now REQUIRED by the scoring
            // engine — the deterministic title/company extractor had ~0%
            // accuracy on varied JD formats, so the API rejects requests
            // that omit them. Validate client-side so the user gets a
            // friendly message instead of a 400 from the API.
            if (!manualCompanyName.trim()) {
                throw new Error("Enter the company name before running JobFit.")
            }
            if (!manualJobTitle.trim()) {
                throw new Error("Enter the job title before running JobFit.")
            }

            if (!accessToken) throw new Error("Please log in again.")

            const data = await postJsonWithDebug(
                `${API_BASE}/api/jobfit`,
                {
                    job: job.trim(),
                    // User-provided labels are required — client-side
                    // validation above has already confirmed both are
                    // non-empty by this point.
                    company_name: manualCompanyName.trim(),
                    job_title: manualJobTitle.trim(),
                    ...(jobUrl.trim() ? { job_url: jobUrl.trim() } : {}),
                    ...(selectedPersonaId
                        ? { persona_id: selectedPersonaId }
                        : {}),
                },
                setDebug,
                accessToken
            )

            const normalized = normalizeJobFitResponseForUI(data)
            setJobFitResult(normalized)

            // The two ids that identify this job from here on. Both come back
            // at the TOP LEVEL of the response, not inside the scoring result,
            // which is why they are read off `normalized` and not off
            // job_signals. NULL is a normal outcome for either: both writes are
            // wrapped server-side in catches that deliberately do not fail the
            // scan, so a destination that needs an id simply is not offered.
            setJobfitRunId(normalized?.jobfit_run_id ?? null)
            setJobfitApplicationId(normalized?.signal_application_id ?? null)
            // A new scan is a new question. Reset the prompt so a company
            // added on the previous scan does not read as this one's answer.
            setNetPromptState("idle")
            setNetPromptError(null)
            setNetPromptCompany(null)

            const ctxFromApi = extractJobContextFromJobFit(normalized)
            setJobContext((prev) => {
                if (ctxFromApi.title || ctxFromApi.company) return ctxFromApi
                return prev
            })

            setTab("jobfit")
        } catch (e: any) {
            setErrorMsg(e?.message || String(e))
        } finally {
            setLoadingJobFit(false)
        }
    }

    // Add this job's company to the networking board and link the tracked
    // application to it, in one server-side call.
    //
    // BY NAME, not by id. The endpoint find-or-creates against
    // uq_network_companies_name (client_profile_id, lower(name)) and treats a
    // unique violation as success, so pressing this for a company already on
    // the board links to the existing row instead of erroring. Every query on
    // that path pins client_profile_id, so this cannot reach another user's
    // board even in principle.
    //
    // JOB TRACKING IS NOT TOUCHED. The application already exists by the time
    // this renders; this is only about the networking board.
    async function addCompanyToNetworkBoard(companyName: string) {
        if (!jobfitApplicationId || netPromptState === "busy") return
        setNetPromptState("busy")
        setNetPromptError(null)
        try {
            if (!accessToken) throw new Error("Please log in again.")
            const data = await postJsonWithDebug(
                `${API_BASE}/api/network/companies/link-application`,
                {
                    application_id: jobfitApplicationId,
                    company_name: companyName,
                },
                setDebug,
                accessToken
            )
            // The board's own spelling wins over the one extracted from the
            // posting: linking to an existing "Globex" from a scan that read
            // "GLOBEX" should confirm the name the user will actually see on
            // their board. Falls back to what we sent if the field is absent.
            setNetPromptCompany(data?.company?.name ?? companyName)
            setNetPromptState("added")
        } catch (e: any) {
            setNetPromptError(e?.message || String(e))
            setNetPromptState("error")
        }
    }

    /* ---------- RENDER: JOBFIT ---------- */
    function renderJobFit() {
        if (!jobFitResult) {
            return null
        }

        // ── Data ─────────────────────────────────────────────────
        const decisionRaw = String(jobFitResult?.decision ?? "").trim()
        const decision = decisionRaw || "REVIEW"
        const lower = decision.toLowerCase()

        const score =
            typeof jobFitResult?.score === "number"
                ? jobFitResult.score
                : Number(jobFitResult?.score ?? 0)
        const hasScore = Number.isFinite(score) && score > 0

        const nextStep = String(jobFitResult?.next_step ?? "").trim()

        // No `whyCodes` read any more: the Why card takes its prose from
        // `bullets` and its keyword from `why_structured`, and the only thing
        // why_codes fed was the deleted client-side re-sort. risk_codes is
        // still read — it is the middle rung of the risk cascade below.
        const riskCodes: any[] = Array.isArray(jobFitResult?.risk_codes)
            ? jobFitResult.risk_codes
            : []

        const renderedBullets: string[] = pickFirstArray(
            jobFitResult?.bullets,
            jobFitResult?.why
        )
            .map(toSafeLine)
            .map((s: string) => s.trim())
            .filter(Boolean)

        // why_structured is the structured form of why bullets from V5.
        // Used only for the keyword heading on each Why card — body text
        // continues to come from the joined `why` string array above so
        // the prose render path stays battle-tested. Per-index lookup at
        // render time; indices align because both arrays are derived from
        // the same `whyBullets` slice in bulletGeneratorV5.ts.
        const whyStructured: any[] = Array.isArray(jobFitResult?.why_structured)
            ? jobFitResult.why_structured
            : []

        const renderedRisks: string[] = pickFirstArray(
            jobFitResult?.risk,
            jobFitResult?.risk_bullets,
            jobFitResult?.risk_flags,
            jobFitResult?.risks
        )
            .map(toSafeLine)
            .map((s: string) => s.trim())
            .filter(Boolean)

        const gate = jobFitResult?.gate_triggered ?? { type: "none" }
        const isForcePass = gate?.type === "force_pass"
        const gateDetail = String(gate?.detail ?? "").trim()

        // Job context is computed once at component scope now — see jobCtx —
        // because the Active Job row on Positioning and Cover Letter shows the
        // same facts and cannot reach inside this function for them.
        const { ctxTitle, ctxCompany, isInternship, locLabel, yearsRequired } =
            jobCtx

        const isPass = lower.includes("pass") || lower.includes("do not")
        const isReview = lower.includes("review")
        const isPriority = lower.includes("priority")

        // ── Decision palette ─────────────────────────────────────
        //
        // ONE COLOUR PER VERDICT, and it is the only thing the verdict varies
        // now. The four gradients this replaces (stripe, score bar, CTA fill,
        // move dot) each encoded the same fact a fifth time; the stripe and the
        // bar are gone, and the CTA is peach on every verdict because peach
        // means "this is the action", not "this is good news".
        //
        // Priority Apply reads as Apply. It used to get its own gradient-filled
        // wordmark, which made the strongest verdict the hardest to read.
        const decisionColor = isPass
            ? L_VERDICT_PASS
            : isReview
              ? L_VERDICT_REVIEW
              : L_VERDICT_APPLY

        // ONE PRIMARY, AND IT IS ALWAYS THE SAME ACTION.
        //
        // This used to fork on the verdict: a Pass swapped the label to "Run
        // New Job" and the handler with it, so the primary button did not
        // proceed at all — it discarded the job. The student who disagreed with
        // a Pass, or who wanted the positioning read precisely BECAUSE the fit
        // was weak, had no route forward that did not start by throwing the
        // scan away. The verdict is advice; it is not a permission check.
        //
        // Review's own wording survives untouched — "Proceed Carefully" already
        // said the cautious thing without removing the action, which is exactly
        // the shape a Pass now gets too. The Pass caveat is a sentence above the
        // row (renderPassNote), not a missing button.
        const ctaLabel = loadingPositioning
            ? "Running Positioning"
            : isReview
              ? "Proceed Carefully"
              : "Proceed to Positioning"

        // The client-side why_codes re-sort (WHY_PRIORITY, weight-first) went
        // with WhyCards, its only consumer. NOTHING ON SCREEN DEPENDED ON IT:
        // the Why card renders `bullets` in the order the API returned them,
        // paired with why_structured[i] by index — and those two arrays are
        // built from the same slice server-side, so a client re-sort could only
        // ever have desynced them. This is unrelated to the WHY-priority
        // ordering inside the scoring engine, which stays where it is.

        // ── Helpers ──────────────────────────────────────────────
        // cleanFactQuote and cleanRiskTitle went with WhyCards / RiskCards —
        // the two card renderers that had not been called since the two-column
        // layout landed. The live risk list does its own normalising inline.

        /**
         * The Pass caveat. Everything the removed gate was trying to say, said
         * in a sentence instead of said by taking the button away.
         *
         * Above the CTA row rather than beside it: the row is two equal-width
         * buttons that reflow to one column on a narrow screen, and a third
         * element inside it would either break that pairing or get pushed onto
         * its own line anyway. Read order is caveat, then action — which is the
         * order a caution belongs in.
         */
        function renderPassNote() {
            if (!isPass) return null
            return (
                <div style={jfPassNote}>
                    SIGNAL scored this a Pass. You can still work it, and the
                    tools below run on any job, but the gaps listed in Risks are
                    the ones a reader will land on first.
                </div>
            )
        }

        // -- Networking prompt ------------------------------------
        // NOW A BUTTON, NOT A STRIP. It used to be a tinted blue band under the
        // verdict card carrying its own sentence ("One person inside beats ten
        // applications"), an Add button and a Not now link.
        //
        // IT IS HALF OF A MATCHED PAIR, NOT A SECONDARY. It spent one pass as
        // the white outlined button beside the peach primary, which was wrong:
        // proceeding to Positioning and adding the company to the tracker are
        // both things we want the student to do, and painting one of them as
        // the quiet option answered a question nobody asked. Both are peach on
        // navy ink now, from one shared style so "same height, same weight,
        // equal width" is structural rather than two lists of numbers that
        // agree today. "Not now" is the quiet option, and it is a text link.
        //
        // UNCHANGED: the three states (idle / busy / added), the dismissal, and
        // the addCompanyToNetworkBoard(ctxCompany) call itself.
        //
        // CHANGED: the Pass suppression is gone. It read "a coach rejected the
        // premise that a rejection is a good moment to network" — but a Pass is
        // SIGNAL's opinion of one posting's fit, not a rejection by the
        // employer, and adding the company to the board is a bet on the
        // company, not on this job. Suppressing it meant the weaker the fit,
        // the less the product would help you get inside — backwards, and the
        // student never saw the option existed. The button is offered on every
        // verdict now; the Pass caveat above the row is what carries the
        // warning, and "Not now" is still one click.
        //
        // ctxCompany rather than a second read of job_signals: it already
        // falls back through job context and the job text, so a cached result
        // with a thin job_signals still names the company.
        const netPromptVisible =
            Boolean(jobfitApplicationId) && Boolean(ctxCompany)

        const showNetButton =
            netPromptVisible && netPromptState !== "dismissed"

        /**
         * The second half of the CTA pair. Renders the "added" confirmation in
         * place rather than vanishing, so the click has a visible result.
         *
         * That confirmation stays white while the button it replaces is peach,
         * and the asymmetry is the point: peach is what to press, and this is
         * a thing already done. Its padding and radius match the pair so the
         * row does not resize under the cursor.
         */
        function renderNetButton() {
            if (!showNetButton) return null

            if (netPromptState === "added") {
                return (
                    <div
                        style={{
                            flex: 1,
                            minWidth: 200,
                            display: "flex",
                            alignItems: "center",
                            gap: 9,
                            padding: "15px 18px",
                            borderRadius: 13,
                            background: L_CARD,
                            border: `1px solid ${L_BORDER}`,
                            fontSize: 13.5,
                            color: L_TEXT_2,
                            lineHeight: "18px",
                            boxSizing: "border-box" as const,
                        }}
                    >
                        <span
                            style={{
                                width: 8,
                                height: 8,
                                borderRadius: "50%",
                                background: L_TEAL,
                                flexShrink: 0,
                            }}
                        />
                        {(netPromptCompany ?? ctxCompany) +
                            " is on your networking board"}
                    </div>
                )
            }

            return (
                <button
                    type="button"
                    disabled={netPromptState === "busy"}
                    onClick={() => addCompanyToNetworkBoard(ctxCompany)}
                    style={{
                        // The same style object the primary uses. Not a copy
                        // of its values — a copy drifts the first time one of
                        // them is touched.
                        ...jfCtaPair,
                        cursor:
                            netPromptState === "busy" ? "default" : "pointer",
                        opacity: netPromptState === "busy" ? 0.6 : 1,
                    }}
                >
                    {netPromptState === "busy"
                        ? "Adding…"
                        : `Add ${ctxCompany} to Network Tracker`}
                </button>
            )
        }

        /**
         * Dismissal and the failure message. These were one function returning
         * both; they are two now because they no longer belong in the same
         * place. "Not now" joins "Run a different job" in the quiet row — they
         * are the same kind of thing, and stacking two lone links looked like
         * two separate afterthoughts. The error is a sentence, so it keeps its
         * own full-width line below and must not be pulled into a flex row.
         *
         * Its own marginTop is gone with the move: jfQuietRow owns the spacing
         * now, and a button carrying its own top margin inside a flex row is
         * how rows end up misaligned by 10px.
         */
        function renderNetDismiss() {
            if (!showNetButton || netPromptState === "added") return null
            return (
                <button
                    type="button"
                    onClick={() => setNetPromptState("dismissed")}
                    style={jfQuietBtn}
                >
                    Not now
                </button>
            )
        }

        function renderNetError() {
            if (!showNetButton || !netPromptError) return null
            return (
                <div
                    style={{
                        fontSize: 12.5,
                        color: "#C0322F",
                        fontWeight: 700,
                        marginTop: 8,
                    }}
                >
                    {netPromptError}
                </div>
            )
        }

        // ── Location label ───────────────────────────────────────
        // ── Risk rows ────────────────────────────────────────────
        // Hoisted out of the JSX so the card can hide when the cascade yields
        // nothing. THE CASCADE AND THE DEDUPE ARE UNCHANGED: risk_structured
        // first (it carries the keyword), then raw riskCodes, then the
        // rendered prose with no keyword at all; then drop any row whose
        // normalised job_fact already appeared.
        //
        // THE BODY IS `gap`, NOT `adjacent_evidence`. It used to be
        // adjacent_evidence alone, which is the secondary support and is
        // explicitly allowed to be an empty string — so every risk without
        // adjacent evidence rendered as a keyword over a blank line, and the
        // gap itself, the actual risk, never reached the screen at all. This
        // is the same join formatRiskBullet does server-side.
        const riskRows = (
            jobFitResult?.risk_structured?.length > 0
                ? jobFitResult.risk_structured.map((r: any) => {
                      const gap = String(r?.gap ?? "").trim()
                      const adjacent = String(r?.adjacent_evidence ?? "").trim()
                      return {
                          risk: adjacent ? gap + " " + adjacent : gap,
                          job_fact: r.keyword,
                          code: r.keyword,
                      }
                  })
                : riskCodes.length > 0
                  ? riskCodes
                  : renderedRisks.map((r: string) => ({
                        risk: r,
                        job_fact: "",
                        code: "",
                    }))
        ).filter((rc: any, idx: number, arr: any[]) => {
            const norm = (v: any) =>
                String(v ?? "")
                    .toLowerCase()
                    .trim()
                    .replace(
                        /^posting (lists?|requires?|calls for|mentions|prefers?)[:\s]*/i,
                        ""
                    )
                    .replace(/\s+as (required|preferred)\.?$/i, "")
                    .replace(/\s+tool usage$/i, "")
                    .replace(/\.?$/, "")
                    .trim()
            const fact = norm(rc?.job_fact)
            if (!fact) return true
            return arr.findIndex((x: any) => norm(x?.job_fact) === fact) === idx
        })

        // Both cards hide when empty — there is no empty state on either any
        // more. The Why card used to render an italic paragraph explaining
        // that no proof matched, on every non-Pass verdict; a card whose only
        // content is an apology for having no content is worse than no card.
        const showWhyCard = renderedBullets.length > 0
        const showRiskCard = riskRows.length > 0
        const hasMoveBlock = Boolean((isForcePass && gateDetail) || nextStep)

        // The verdict word. Priority Apply keeps its name — it is a different
        // verdict and the tracker records it as one — but it is no longer a
        // two-line gradient wordmark, which made the strongest result the
        // hardest one to read.
        const verdictWord = isPass
            ? "Pass"
            : isReview
              ? "Review"
              : isPriority
                ? "Priority Apply"
                : "Apply"

        // Context line parts, joined by dots. Same five facts the pill row
        // carried; a title with no company still reads correctly because the
        // empties are filtered before the join.
        // ── Full render ──────────────────────────────────────────
        return (
            <>
                {/* Job context. Title and company lead at 17px/700; the
                    qualifiers sit under them. It was one 13px muted line with
                    all five facts dot-joined, which made the two that identify
                    the job the same weight as "3+ yrs exp". */}
                <JobContextBlock ctx={jobCtx} />

                {/* Verdict card — decision, score and the move, in one navy
                    block. Was three: a stripe, a hero split by a rule, and a
                    separately-bordered "Your move" strip. */}
                <div style={jfVerdictCard}>
                    <div style={jfVerdictTop}>
                        <div>
                            <div style={jfVerdictEyebrow}>Signal decision</div>
                            <div
                                style={{
                                    ...jfVerdictWord,
                                    color: decisionColor,
                                }}
                            >
                                {verdictWord}
                            </div>
                        </div>
                        {hasScore ? (
                            <div style={jfScoreWrap}>
                                <div style={jfScoreNum}>{score}</div>
                                <div style={jfScoreOf}>/100</div>
                            </div>
                        ) : null}
                    </div>

                    {hasMoveBlock ? (
                        <>
                            <div style={jfVerdictDivider} />
                            <div style={jfVerdictEyebrow}>Your move</div>
                            {isForcePass && gateDetail ? (
                                <div style={jfMoveBody}>{gateDetail}</div>
                            ) : null}
                            {nextStep ? (
                                <div
                                    style={{
                                        ...jfMoveBody,
                                        marginTop:
                                            isForcePass && gateDetail ? 10 : 0,
                                    }}
                                >
                                    {nextStep}
                                </div>
                            ) : null}
                        </>
                    ) : null}
                </div>

                {/* Actions. The Pass caveat, then a matched pair from
                    jfCtaPair — see renderNetButton — then the quiet row.

                    Both buttons render on every verdict now. The primary goes
                    straight to proceedToPositioning: the onPrimaryClick wrapper
                    it used to call existed only to branch on isPass, and with
                    that branch gone it was a function whose whole body was a
                    call to another function. */}
                {renderPassNote()}
                <div style={jfCtaRow}>
                    <button
                        style={jfCtaPair}
                        onClick={proceedToPositioning}
                        disabled={loadingPositioning}
                        type="button"
                    >
                        {loadingPositioning ? (
                            <span style={spinnerOnNavy} aria-hidden />
                        ) : null}
                        {ctaLabel}
                    </button>
                    {renderNetButton()}
                </div>
                {/* Run New Job, demoted. On a Pass it used to BE the primary,
                    which is how discarding the scan ended up as the loudest
                    thing on the screen. It is still one click from here, and it
                    is still in the side nav under Actions, but it no longer
                    occupies the slot that moving forward belongs in. Shares the
                    row and the quiet treatment with "Not now" beside it. */}
                <div style={jfQuietRow}>
                    <button
                        type="button"
                        onClick={() => runNewJob()}
                        style={jfQuietBtn}
                    >
                        Run a different job
                    </button>
                    {renderNetDismiss()}
                </div>
                {renderNetError()}

                {/* Why / Risk. Stacked full width, Why first. They were side by
                    side in a 1fr 1fr grid, which gave each of them half a
                    column for prose that runs two and three lines — the two
                    cards are a sequence, not a comparison, so they read down
                    the page now instead of across it. */}
                {showWhyCard || showRiskCard ? (
                    <div style={jfPanelStack}>
                        {showWhyCard ? (
                            <div style={jfPanel}>
                                {/* THE ONE TITLE THAT VARIES. On a Pass these
                                    strengths are not why you are competitive
                                    for this job — we have just told the student
                                    to skip it — they are what to carry to the
                                    next one. Calling them competitive here
                                    would be a claim the verdict directly
                                    contradicts. The risk title does NOT vary:
                                    the force-pass wording it used to carry
                                    ("Why This Is a Pass") is redundant now that
                                    the gate reason renders in YOUR MOVE. */}
                                <div style={{ ...jfPanelRule, background: L_TEAL }} />
                                <div style={jfPanelHeader}>
                                    <div style={sectionLabel}>
                                        {isPass
                                            ? "Strengths to remember"
                                            : "Why you are competitive"}
                                    </div>
                                    <div
                                        style={{
                                            ...jfPanelHeaderCount,
                                            color: L_BLUE,
                                        }}
                                    >
                                        {renderedBullets.length}{" "}
                                        {renderedBullets.length === 1
                                            ? "signal"
                                            : "signals"}
                                    </div>
                                </div>
                                <div style={jfPanelBody}>
                                    {renderedBullets.map(
                                        (bullet: string, i: number) => {
                                            // Keyword heading source — V5
                                            // why_structured[i].keyword. Pre-V5
                                            // cached results have no structured
                                            // array; the empty string
                                            // suppresses the slot.
                                            const keyword = sentenceCase(
                                                String(
                                                    whyStructured[i]?.keyword ||
                                                        ""
                                                ).trim()
                                            )
                                            const parts = splitAdvice(
                                                bullet,
                                                whyStructured[i]
                                            )
                                            return (
                                                <div
                                                    key={i}
                                                    style={
                                                        i ===
                                                        renderedBullets.length -
                                                            1
                                                            ? jfBulletLast
                                                            : jfBullet
                                                    }
                                                >
                                                    {keyword ? (
                                                        <div
                                                            style={
                                                                jfBulletKeyword
                                                            }
                                                        >
                                                            {keyword}
                                                        </div>
                                                    ) : null}
                                                    <div style={jfBulletProse}>
                                                        {parts.observation}
                                                    </div>
                                                    {parts.advice ? (
                                                        <div
                                                            style={{
                                                                ...jfAdvice,
                                                                background:
                                                                    L_WHY_ADVICE_BG,
                                                            }}
                                                        >
                                                            <div
                                                                style={{
                                                                    ...jfAdviceEyebrow,
                                                                    color: L_WHY_ADVICE_EYEBROW,
                                                                }}
                                                            >
                                                                Do this
                                                            </div>
                                                            <div
                                                                style={{
                                                                    ...jfAdviceBody,
                                                                    color: L_WHY_ADVICE_INK,
                                                                }}
                                                            >
                                                                {parts.advice}
                                                            </div>
                                                        </div>
                                                    ) : null}
                                                </div>
                                            )
                                        }
                                    )}
                                </div>
                            </div>
                        ) : null}

                        {showRiskCard ? (
                            <div style={jfPanel}>
                                <div style={{ ...jfPanelRule, background: L_RED }} />
                                <div style={jfPanelHeader}>
                                    <div style={sectionLabel}>
                                        Your risks
                                    </div>
                                    <div
                                        style={{
                                            ...jfPanelHeaderCount,
                                            color: L_RED,
                                        }}
                                    >
                                        {riskRows.length}{" "}
                                        {riskRows.length === 1 ? "gap" : "gaps"}
                                    </div>
                                </div>
                                <div style={jfPanelBody}>
                                    {riskRows.map((rc: any, i: number) => {
                                        const jobFact = String(
                                            rc?.job_fact ?? ""
                                        ).trim()
                                        const riskText = String(
                                            rc?.risk ?? rc ?? ""
                                        ).trim()
                                        const riskCode = String(rc?.code ?? "")

                                        // Code-based keyword fallbacks when
                                        // job_fact is empty. Sentence case now,
                                        // like every other keyword on the card.
                                        const codeFallbacks: Record<
                                            string,
                                            string
                                        > = {
                                            RISK_EXPERIENCE:
                                                "Experience requirement",
                                            RISK_SENIORITY_MISMATCH:
                                                "Seniority level",
                                            RISK_MISSING_PROOF: "Missing proof",
                                            RISK_LOCATION:
                                                "Location constraint",
                                            RISK_CONTRACT: "Contract role",
                                            RISK_SALES: "Sales expectations",
                                            RISK_ANALYTICS_HEAVY:
                                                "Analytics requirement",
                                            RISK_OVERQUALIFIED: "Overqualified",
                                        }

                                        const keywordFromFact = jobFact
                                            ? jobFact
                                                  .replace(
                                                      /^posting (lists?|requires?|calls for|mentions|prefers?|suggests?)[:\s]*/i,
                                                      ""
                                                  )
                                                  .replace(
                                                      /\s+as (required|preferred|a requirement)\.?$/i,
                                                      ""
                                                  )
                                                  .replace(/\.?$/, "")
                                                  .trim()
                                            : ""
                                        const keyword = sentenceCase(
                                            keywordFromFact ||
                                                codeFallbacks[riskCode] ||
                                                ""
                                        )
                                        // NO ADVICE BLOCK ON THIS CARD. The Why
                                        // side has one because V5 ships an
                                        // `action` per bullet; RiskBullet is
                                        // {keyword, gap, adjacent_evidence,
                                        // severity} with no advice field and
                                        // no arrow in its formatter, so there
                                        // is nothing to render and inventing
                                        // one is not on offer. Risks are
                                        // observation-only until that field
                                        // exists server-side.
                                        return (
                                            <div
                                                key={i}
                                                style={
                                                    i === riskRows.length - 1
                                                        ? jfBulletLast
                                                        : jfBullet
                                                }
                                            >
                                                {keyword ? (
                                                    <div style={jfBulletKeyword}>
                                                        {keyword}
                                                    </div>
                                                ) : null}
                                                <div style={jfBulletProse}>
                                                    {riskText}
                                                </div>
                                            </div>
                                        )
                                    })}
                                </div>
                            </div>
                        ) : null}
                    </div>
                ) : null}
            </>
        )
    }
