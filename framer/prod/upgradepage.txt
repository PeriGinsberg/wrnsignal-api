/**
 * SIGNAL Upgrade Page — verdict-driven, personalized
 *
 *   - Reads signal_upgrade_handoff from sessionStorage on mount

 *   - Renders one of three variants based on result.decision:
 *       APPLY / PRIORITY APPLY → APPLY variant
 *       REVIEW                → REVIEW variant

 *   - Personalizes meta header from result.job_signals + result.score +

 *     route to Stripe with the email pre-filled. Meta InitiateCheckout +
 *     Google Ads conversion fire before redirect.
 *   - Falls back to a generic APPLY-variant view (no personalized meta)
 *     when no handoff exists (deep link, stale tab).
 *
 * @framerSupportedLayoutWidth any
 * @framerSupportedLayoutHeight any
 */

function ss(key: string): string {
    try {
        return sessionStorage.getItem(key) || ""
    } catch {
        return ""
    }
}

function getCookie(name: string): string {
    if (typeof document === "undefined") return ""
    const m = document.cookie.match(
        new RegExp("(?:^|; )" + name.replace(/[-.]/g, "\\$&") + "=([^;]*)")
    )
    return m ? decodeURIComponent(m[1]) : ""
}

function getAttributionSnapshot() {
    return {
        utm_source: ss("signal_utm_source"),
        utm_medium: ss("signal_utm_medium"),
        utm_campaign: ss("signal_utm_campaign"),
        utm_content: ss("signal_utm_content"),
        utm_term: ss("signal_utm_term"),
        landing_page: ss("signal_landing_page"),
        referrer: ss("signal_referrer"),
        fbclid: ss("signal_fbclid"),
        ttclid: ss("signal_ttclid"),
        gclid: ss("signal_gclid"),
        fbp: getCookie("_fbp"),
        fbc: getCookie("_fbc"),
        ttp: getCookie("_ttp"),
    }
}

// Server-side mirror of the browser pixel. Same event_id rides both
// fbq's eventID and the server CAPI call so Meta dedups the pair inside
// its 7-day (event_name, event_id) window.
type FunnelEventName = "Lead" | "ViewContent" | "AddToCart" | "InitiateCheckout"
type FunnelCustomData = {
    value?: number
    currency?: string
    verdict?: string
    score?: number | null
    cta_source?: string
    content_name?: string
    content_category?: string
}

function genEventId(): string {
    try {
        if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
            return crypto.randomUUID()
        }
    } catch {}
    return (
        Date.now().toString(36) +
        "-" +
        Math.random().toString(36).slice(2, 10)
    )
}

function sendServerEvent(args: {
    event_name: FunnelEventName
    event_id: string
    email?: string
    custom_data?: FunnelCustomData
}) {
    try {
        const sessionId =
            (typeof localStorage !== "undefined" &&
                localStorage.getItem("jobfit_session_id")) ||
            ""
        const attribution = getAttributionSnapshot()
        const body = {
            event_name: args.event_name,
            event_id: args.event_id,
            event_time_sec: Math.floor(Date.now() / 1000),
            email: args.email || undefined,
            session_id: sessionId,
            ...attribution,
            custom_data: args.custom_data || {},
        }
        const url = API_BASE + "/api/conversions/event"
        // fetch with keepalive:true + credentials:"omit". sendBeacon would
        // be simpler but forces credentials:"include" which requires the
        // API to enable Access-Control-Allow-Credentials globally — not
        // wanted on a bearer-token-auth API.
        fetch(url, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
            keepalive: true,
            credentials: "omit",
        }).catch(() => {})
    } catch {}
}

function isValidEmail(s: string): boolean {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(s.trim())
}

function decisionToVariantKey(decision: string | undefined | null): VariantKey {
    const d = String(decision || "").trim().toUpperCase()
    if (d === "REVIEW") return "review"
    if (d === "PASS") return "pass"
    return "apply"
}
