function ss(key: string): string {
    try {
        return sessionStorage.getItem(key) || ""
    } catch {
        return ""
    }
}

function getCookie(name: string): string {
    if (typeof document === "undefined") return ""
    const m = document.cookie.match(
        new RegExp("(?:^|; )" + name.replace(/[-.]/g, "\\$&") + "=([^;]*)")
    )
    return m ? decodeURIComponent(m[1]) : ""
}

// Server-side mirror of the browser pixel. Same event_id rides both
// fbq's eventID and the server CAPI call so Meta dedups the pair inside
// its 7-day (event_name, event_id) window.
type FunnelEventName = "Lead" | "ViewContent" | "AddToCart" | "InitiateCheckout"
type FunnelCustomData = {
    value?: number
    currency?: string
    verdict?: string
    score?: number | null
    cta_source?: string
    content_name?: string
    content_category?: string
}

function genEventId(): string {
    try {
        if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
            return crypto.randomUUID()
        }
    } catch {}
    return (
        Date.now().toString(36) +
        "-" +
        Math.random().toString(36).slice(2, 10)
    )
}

function sendServerEvent(args: {
    event_name: FunnelEventName
    event_id: string
    email?: string
    custom_data?: FunnelCustomData
}) {
    try {
        const sessionId =
            (typeof localStorage !== "undefined" &&
                localStorage.getItem("jobfit_session_id")) ||
            ""
        const attribution = getAttributionSnapshot()
        const body = {
            event_name: args.event_name,
            event_id: args.event_id,
            event_time_sec: Math.floor(Date.now() / 1000),
            email: args.email || undefined,
            session_id: sessionId,
            ...attribution,
            custom_data: args.custom_data || {},
        }
        const url = API_BASE + "/api/conversions/event"
        // fetch with keepalive:true + credentials:"omit". sendBeacon would
        // be simpler but forces credentials:"include" which requires the
        // API to enable Access-Control-Allow-Credentials globally — not
        // wanted on a bearer-token-auth API.
        fetch(url, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
            keepalive: true,
            credentials: "omit",
        }).catch(() => {})
    } catch {}
}

function isValidEmail(s: string): boolean {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(s.trim())
}

function decisionToVariantKey(decision: string | undefined | null): VariantKey {
    const d = String(decision || "").trim().toUpperCase()
    if (d === "REVIEW") return "review"
    if (d === "PASS") return "pass"
    return "apply"
}

            sessionStorage.removeItem(HANDOFF_KEY)
            const parsed = JSON.parse(raw) as HandoffPayload
            const ts = typeof parsed.ts === "number" ? parsed.ts : 0
            const fresh = ts && Date.now() - ts < HANDOFF_MAX_AGE_MS
            if (fresh && parsed.result) {
                setResult(parsed.result)
                if (parsed.cta_source) setCtaSource(parsed.cta_source)
            }
        } catch {}
        setHydrated(true)

        const sessionId = localStorage.getItem("jobfit_session_id") || crypto.randomUUID()
        if (!localStorage.getItem("jobfit_session_id")) {
            localStorage.setItem("jobfit_session_id", sessionId)
        }
        fetch(API_BASE + "/api/track", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                session_id: sessionId,
                page_path: window.location.pathname || "/signal/upgrade",
                page_name: "upgrade_page",
                referrer: document.referrer || null,
                utm_source: ss("signal_utm_source") || null,
                utm_medium: ss("signal_utm_medium") || null,
                utm_campaign: ss("signal_utm_campaign") || null,
            }),
        }).catch(() => {})
    }, [])

    // Fire GA4 upgrade_page_viewed once after handoff hydration. Meta
    // PageView is auto-fired by the Framer base pixel script. Includes
    // verdict + score + cta_source so the funnel can be sliced by which
    // result the user is upgrading from. Fires once per mount regardless
    // of whether result hydrated successfully.
    const viewFiredRef = React.useRef(false)
    React.useEffect(() => {
        if (!hydrated || viewFiredRef.current) return
        viewFiredRef.current = true
        const verdict = result ? String(result.decision || "").toUpperCase() : null
        const score =
            result && typeof result.score === "number" ? result.score : null
        const utm = getEventUtmParams()
        try {
            if (typeof window !== "undefined" && (window as any).gtag) {
                (window as any).gtag("event", "upgrade_page_viewed", {
                    event_category: "funnel",
                    verdict,
                    score,
                    cta_source: ctaSource || null,
                    has_handoff: !!result,
                    ...utm,
                })
            }
        } catch {}
    }, [hydrated, result, ctaSource])

    const variantKey = decisionToVariantKey(result?.decision)
    const variant = VARIANTS[variantKey]

    const displayDecision = String(result?.decision || "").trim().toUpperCase() || "APPLY"
    const displayScore = typeof result?.score === "number" ? result.score : null
    const displayJobTitle = result?.job_signals?.jobTitle || "this role"
    const displayCompany = result?.job_signals?.companyName || "the company"
    const displayMove = (result?.next_step && result.next_step.trim()) || variant.defaultMove

    async function submitCheckout(
        email: string,
        source: string,
        setError: (s: string) => void,
        setSubmitting: (b: boolean) => void
    ) {
        const trimmed = email.trim().toLowerCase()
        setError("")
        if (!isValidEmail(trimmed)) {
            setError("Please enter a valid email address.")
            return
        }
        setSubmitting(true)
        try {
            const res = await fetch(`${API_BASE}/api/checkout/create-session`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    email: trimmed,
                    source,
                    // Correlation key back to the original anonymous trial scan,
                    // captured server-side into unlock_email_captures.session_id
                    // so the funnel scan→unlock→purchase chain is joinable.
                    // Stashed in localStorage by /api/jobfit-run-trial-open.
                    session_id:
                        localStorage.getItem("jobfit_session_id") || undefined,
                    ...getAttributionSnapshot(),
                }),
            })
            const data = await res.json()
            if (!data?.url) {
                throw new Error(data?.error || "Checkout failed. Please try again.")
            }
            const verdict = result
                ? String(result.decision || "").toUpperCase()
                : null
            const score =
                result && typeof result.score === "number" ? result.score : null
            const utm = getEventUtmParams()

            const initEventId = genEventId()
            try {
                if (typeof window !== "undefined" && (window as any).fbq) {
                    (window as any).fbq(
                        "track",
                        "InitiateCheckout",
                        {
                            value: 99,
                            currency: "USD",
                            content_name: "SIGNAL 90-day access",
                            verdict,
                            score,
                            ...utm,
                        },
                        { eventID: initEventId }
                    )
                }
            } catch {}
            sendServerEvent({
                event_name: "InitiateCheckout",
                event_id: initEventId,
                email: trimmed || undefined,
                custom_data: {
                    value: 99,
                    currency: "USD",
                    content_name: "SIGNAL 90-day access",
                    verdict: verdict || undefined,
                    score,
                },
            })
            try {
                if (typeof window !== "undefined" && (window as any).gtag) {
                    (window as any).gtag("event", "checkout_initiated", {
                        event_category: "funnel",
                        verdict,
                        score,
                        ...utm,
                    })
                }
            } catch {}
            try {
                if (typeof window !== "undefined" && (window as any).gtag) {
                    (window as any).gtag("event", "conversion", {
                        send_to: "AW-11125129027/AMw8CMHsraUcEMP-77gp",
                        value: 99.0,
                        currency: "USD",
                    })
                }
            } catch {}

            window.open(data.url, "_blank")
            setSubmitting(false)
        } catch (err: any) {
            setError(err?.message || "Something went wrong. Please try again.")
            setSubmitting(false)
        }
    }

    function handleInlineSubmit(e: React.FormEvent<HTMLFormElement>) {
        e.preventDefault()
        const source = ctaSource ? `${ctaSource}__inline` : "upgrade_inline"
        void submitCheckout(inlineEmail, source, setInlineError, setSubmittingInline)
    }

    function handleBottomSubmit(e: React.FormEvent<HTMLFormElement>) {
        e.preventDefault()
        const source = ctaSource ? `${ctaSource}__checkout` : "upgrade_checkout"
        void submitCheckout(bottomEmail, source, setBottomError, setSubmittingBottom)
    }

    function handleReturn(e: React.MouseEvent<HTMLAnchorElement>) {
        e.preventDefault()
        if (typeof window !== "undefined" && window.top) {
            window.top.location.replace(TRIAL_RESULT_URL)
        } else {
            window.location.replace(TRIAL_RESULT_URL)
        }
    }

    function handleNewScan(e: React.MouseEvent<HTMLAnchorElement>) {
        e.preventDefault()
        if (typeof window !== "undefined" && window.top) {
            window.top.location.replace(NEW_LANDING_URL)
        } else {
            window.location.replace(NEW_LANDING_URL)
        }
    }

    if (!hydrated) {
        return (
            <div style={{ width: "100%", minHeight: "100vh", background: CREAM, ...props.style }}>
                <PageShell />
            </div>
        )
    }

    // Presentation-only maps derived from the selected variant.
    const accentColor =
        variant.accent === "green" ? GREEN : variant.accent === "peach" ? PEACH_DEEP : WRN_BLUE_DEEP
    const stepNumBg =
        variant.accent === "green" ? GREEN : variant.accent === "peach" ? PEACH : WRN_BLUE
    const chip =
        variant.chipTone === "green"
            ? { bg: "rgba(29,158,117,0.14)", fg: GREEN }
            : variant.chipTone === "peach"
              ? { bg: "rgba(242,150,59,0.16)", fg: PEACH_DEEP }
              : { bg: "rgba(138,147,161,0.16)", fg: INK }
    const bodyEyebrowColor =
        variant.accent === "green" ? GREEN : variant.accent === "peach" ? PEACH_DEEP : WRN_BLUE_DEEP

    return (
        <div className="up-page" style={{ width: "100%", ...props.style }}>
            <PageShell />

            {/* ===== HEADER ===== */}
            <header className="up-header">
                {/* Select the uploaded navy SIGNAL PNG in the "Logo" image control. */}
                {props.logo ? (
                    <img src={props.logo} alt="SIGNAL" className="up-logo-img" />
                ) : null}
                <a href={LANDING_URL} className="up-login">Log in</a>
            </header>

            {/* ===== HERO (per variant) ===== */}
            <section className="up-hero">
                <div className="up-blob up-blob-a" aria-hidden="true" />
                <div className="up-blob up-blob-b" aria-hidden="true" />
                <div className="up-hero-inner">
                    <div className="up-eyebrow">{variant.heroEyebrow}</div>
                    <h1 className="up-h1">
                        {variant.heroHeadlineLead}
                        <span style={{ color: accentColor }}>{variant.heroHeadlineAccent}</span>
                        {variant.heroHeadlineTail}
                    </h1>
                    <p className="up-sub">{variant.heroSub}</p>

                    <div className="up-chip" style={{ background: chip.bg, color: chip.fg }}>
                        {variant.chipLabel}
                        {variant.chipShowScore && displayScore !== null ? ` · ${displayScore}` : ""}
                    </div>
                    <p className="up-chip-for">for {displayJobTitle}</p>

                    <form
                        onSubmit={handleInlineSubmit}
                        className="up-emailrow"
                        style={{ flexDirection: isMobile ? "column" : "row" }}
                    >
                        <input
                            type="email"
                            className="up-email"
                            placeholder="you@school.edu"
                            value={inlineEmail}
                            onChange={(e) => setInlineEmail(e.target.value)}
                            aria-label="Email address"
                        />
                        <button
                            type="submit"
                            className="up-unlock"
                            disabled={submittingInline}
                        >
                            {submittingInline ? "Starting…" : "Unlock · $99 →"}
                        </button>
                    </form>
                    {inlineError ? <p className="up-error">{inlineError}</p> : null}
                    <p className="up-fine">One-time. Unlimited jobs. No subscription.</p>

                    <p className="up-nudge">{variant.heroNudge}</p>
                </div>
            </section>

            {/* ===== BODY (per variant) ===== */}
            <section className="up-body">
                <div className="up-body-inner">
                    <div className="up-body-eyebrow" style={{ color: bodyEyebrowColor }}>
                        {variant.bodyIntroEyebrow}
                    </div>
                    <h2 className="up-body-headline">{variant.bodyIntroHeadline}</h2>
                    {variant.bodyIntroBody ? (
                        <p className="up-body-intro">{variant.bodyIntroBody}</p>
                    ) : (
                        <div style={{ height: 8 }} />
                    )}

                    <div className="up-steps">
                        {variant.steps.map((s) => (
                            <div className="up-step" key={s.n}>
                                <div className="up-step-num" style={{ background: stepNumBg }}>
                                    {s.n}
                                </div>
                                <div>
                                    <div className="up-step-label" style={{ color: bodyEyebrowColor }}>
                                        {s.label}
                                    </div>
                                    <p className="up-step-title">{s.title}</p>
                                </div>
                            </div>
                        ))}
                    </div>
                </div>
            </section>

            {/* ===== DARK HONEST-MATH BEAT (per variant) ===== */}
            <section className="up-math">
                <div className="up-math-inner">
                    <div className="up-math-eyebrow">{variant.mathEyebrow}</div>
                    <p className="up-math-text">{variant.mathText}</p>
                </div>
            </section>

            {/* ===== SHARED CLOSE ===== */}
            <section className="up-close">
                <div className="up-close-inner">
                    <h2 className="up-close-headline">{variant.closeHeadline}</h2>
                    <p className="up-toollist">Job Fit. Positioning. Cover Letter. Networking.</p>

                    <form
                        onSubmit={handleBottomSubmit}
                        className="up-emailrow"
                        style={{ flexDirection: isMobile ? "column" : "row" }}
                    >
                        <input
                            type="email"
                            className="up-email"
                            placeholder="you@school.edu"
                            value={bottomEmail}
                            onChange={(e) => setBottomEmail(e.target.value)}
                            aria-label="Email address"
                        />
                        <button
                            type="submit"
                            className="up-unlock"
                            disabled={submittingBottom}
                        >
                            {submittingBottom ? "Starting…" : "Unlock · $99 →"}
                        </button>
                    </form>
                    {bottomError ? <p className="up-error">{bottomError}</p> : null}
                    <p className="up-fine">One-time. Unlimited jobs. No subscription.</p>

                    <p className="up-crossplat">Web and mobile. Your unlock works on both.</p>
                    <a
                        href={APP_STORE_URL}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="up-badge"
                        aria-label="Download SIGNAL on the App Store"
                    >
                        <img
                            src="https://tools.applemediaservices.com/api/badges/download-on-the-app-store/black/en-us"
                            alt="Download on the App Store"
                        />
                    </a>

                    <p className="up-guarantee">
                        7-day money-back guarantee. If SIGNAL doesn't change how you search, we refund every dollar.
                    </p>

                    <div className="up-backlinks">
                        <a href={TRIAL_RESULT_URL} className="up-backlink" onClick={handleReturn}>
                            ← Back to my results
                        </a>
                        <a href={NEW_LANDING_URL} className="up-backlink" onClick={handleNewScan}>
                            Run a new scan →
                        </a>
                    </div>
                </div>
            </section>
        </div>
    )
}
