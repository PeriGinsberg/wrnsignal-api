 *           plus locked teasers for cover letter, networking, resume

 *   - { status: "ok", locked: true, lock_reason: "free_run_used" }

 * @framerSupportedLayoutHeight any

const API_BASE = "https://wrnsignal-api.vercel.app"

// Public landing destination for users who arrive without an email param.
// This page expects the email to come pre-filled from the landing page CTA;
// when it doesn't, we route them back so they enter their email there.
const LANDING_FALLBACK = "https://wrnsignal.workforcereadynow.com/signal"

// Minimum content lengths enforced server-side. Mirror them client-side so
// the user gets a friendly message before we send a doomed request.
const MIN_RESUME_CHARS = 100
const MIN_JD_CHARS = 100

// sessionStorage handoff from the v2 landing page (landingpage.txt) — the
// landing form extracts the resume PDF and stashes {resume_text,
// job_description, attribution, ts} here, then navigates to this page. When
// present we hydrate state and auto-run against /api/jobfit-run-trial-open
// (no email, no credit gate), skipping the manual input view entirely.
const OPEN_FLOW_HANDOFF_KEY = "signal_open_trial_handoff"

// Max age for a handoff payload to be honored. Beyond this we ignore it
// (stale tab, user navigated back hours later) and show the input view.
const OPEN_FLOW_HANDOFF_MAX_AGE_MS = 10 * 60 * 1000 // 10 minutes

// Non-destructive peek at the v2 landing handoff. Mirrors the read in the
// mount effect exactly (same key, parse, ts, freshness, min-length checks)
// but never removes the key. Used by the loading initializer so a handoff
// entry starts on the loader instead of flashing the input view, and re-used
// by the effect, which stays the sole owner of removeItem + the auto-run.
// Returns null when absent, stale, invalid, or storage is unavailable
// (SSR / blocked).
function peekFreshOpenFlowHandoff(): { resume: string; jd: string } | null {
    if (typeof window === "undefined") return null
    try {
        const raw = sessionStorage.getItem(OPEN_FLOW_HANDOFF_KEY)
        if (!raw) return null
        const parsed = JSON.parse(raw) as {
            resume_text?: string
            job_description?: string
            ts?: number
        }
        const ts = typeof parsed.ts === "number" ? parsed.ts : 0
        const fresh = ts && Date.now() - ts < OPEN_FLOW_HANDOFF_MAX_AGE_MS
        const r = String(parsed.resume_text || "").trim()
        const j = String(parsed.job_description || "").trim()
        if (fresh && r.length >= MIN_RESUME_CHARS && j.length >= MIN_JD_CHARS) {
            return { resume: r, jd: j }
        }
        return null
    } catch {
        return null
    }
}

// Resume carryover for Start Over flow. When the user clicks Start Over

// new landing instead. Relative "/" keeps the user on whichever domain

function AppStoreBadge() {
    const [isMobile, setIsMobile] = React.useState(false)
    React.useEffect(() => {
        if (typeof window === "undefined") return
        const check = () => setIsMobile(window.innerWidth < 768)
        check()
        window.addEventListener("resize", check)
        return () => window.removeEventListener("resize", check)
    }, [])
    // Badge sits below the dev banner. Prod has no banner, so offset is 0.
    const bannerOffset = DEV_MODE ? 40 : 0
    return (
        <a
            href={APP_STORE_URL}
            target="_blank"
            rel="noopener noreferrer"
            aria-label="Download SIGNAL on the App Store"
            style={{
                position: "fixed" as const,
                top: (isMobile ? 16 : 24) + bannerOffset,
                right: isMobile ? 16 : 32,
                zIndex: 9999,
                display: "block",
                filter: "drop-shadow(0 4px 12px rgba(0,0,0,0.25))",
            }}
        >
            <img
                src="https://tools.applemediaservices.com/api/badges/download-on-the-app-store/black/en-us"
                alt="Download on the App Store"
                width={isMobile ? 108 : 135}
                height={isMobile ? 32 : 40}
                style={{ display: "block" }}
            />
        </a>
    )
}

// TYPES (mirror /api/jobfit-run-trial response)

type WhyBullet = {
    keyword: string
    lead: string
    connection: string
    action: string
}

type RiskBullet = {
    keyword: string
    gap: string
    adjacent_evidence: string
    severity: "low" | "medium" | "high"
}

type CoverLetterStrategy = {
    open_with: string
    lead_signal: string
    address_gap: string | null
    tone: string
}

type NetworkingStrategy = {
    target_contacts: string
    outreach_angle: string
}

// Free-scan perception (open-flow scans only). Produced by the engine's
// freeScanPerception bridge; rides on result.perception. Drives the new
// handwriting results UI. Absent on gated-flow scans and on fail-open.
type PerceptionMark = {
    line: string
    mark: "underline" | "circle"
    tone: "positive" | "problem"
}
type PerceptionNote = {
    text: string
    tone: "positive" | "problem" | "neutral"
}
type FreeScanPerception = {
    archetype: "STRONG_FIT" | "MISPOSITIONED" | "WRONG_FIELD"
    strengths: string[]
    perception: string
    marked_lines: PerceptionMark[]
    margin_notes: PerceptionNote[]
    verdict_line: string
    cta_lead: string
}

type TrialResult = {
    decision: string
    score: number
    icon?: string
    next_step?: string
    why_structured?: WhyBullet[]
    risk_structured?: RiskBullet[]
    cover_letter_strategy?: CoverLetterStrategy | null
    networking_strategy?: NetworkingStrategy | null
    perception?: FreeScanPerception
}

type LockEntry = { label: string; stripe_url: string }
type Locks = {
    cover_letter: LockEntry
    networking: LockEntry
    resume_positioning: LockEntry
    tracker: LockEntry
    run_another_job: LockEntry
}
type UpgradeBlock = {
    price: string
    term: string
    guarantee: string
    stripe_url: string
}

type TrialResponse =
    | {
          status: "ok"
          locked: boolean
          lock_reason: "none" | "free_run_used"
          result: TrialResult
          locks: Locks
          upgrade: UpgradeBlock
      }
    | {
          status: "out_of_credits"
          locked: true
          lock_reason: "free_run_used"
          result: null
          locks: Locks
          upgrade: UpgradeBlock
      }
    | { ok: false; error: string; detail?: string } // 4xx/5xx error envelope

// MAIN
// ============================================================================

export default function SignalTrialJobFit() {
    // ── Inputs ──────────────────────────────────────────────────────
    const [email, setEmail] = React.useState("")
    const [resume, setResume] = React.useState("")
    const [jd, setJd] = React.useState("")

    // ── State ───────────────────────────────────────────────────────
    const [loading, setLoading] = React.useState(
        () => peekFreshOpenFlowHandoff() !== null
    )
    const [error, setError] = React.useState("")
    const [response, setResponse] = React.useState<TrialResponse | null>(null)
    const [upgradeLoading, setUpgradeLoading] = React.useState(false)
    // True when this page was entered via the v2 landing-page handoff.
    // In that mode the input view never renders; we go straight from
    // loader to result (or error fallback shows the input view).
    const [isOpenFlow, setIsOpenFlow] = React.useState(false)

    // ── Core analysis runner — explicit params, no state reads ───────
    //
    // Extracted so the mount-time auto-run path can call it before React
    // has flushed the state hydrated from the handoff payload. The button-
    // click path (handleAnalyze) also routes through here.
    async function runAnalysis(opts: {
        email: string
        resume: string
        jd: string
        useOpenEndpoint: boolean
    }) {
        const { email: e, resume: r, jd: j, useOpenEndpoint } = opts
        setError("")

        if (!useOpenEndpoint) {
            if (!e || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(e)) {
                setError("Enter your email so we can save your analysis.")
                return
            }
        }
        if (r.length < MIN_RESUME_CHARS) {
            setError(
                `Resume text is too short — need at least ${MIN_RESUME_CHARS} characters.`
            )
            return
        }
        if (j.length < MIN_JD_CHARS) {
            setError(
                `Job description is too short — need at least ${MIN_JD_CHARS} characters.`
            )
            return
        }

        setLoading(true)

        try {
            const sessionId =
                localStorage.getItem("jobfit_session_id") || undefined
            const endpoint = useOpenEndpoint
                ? "/api/jobfit-run-trial-open"
                : "/api/jobfit-run-trial"
            const body: Record<string, unknown> = {
                resume_text: r,
                job_description: j,
                session_id: sessionId,
                utm_source: ss("signal_utm_source") || undefined,
                utm_medium: ss("signal_utm_medium") || undefined,
                utm_campaign: ss("signal_utm_campaign") || undefined,
            }
            if (!useOpenEndpoint) body.email = e

            const res = await fetch(API_BASE + endpoint, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(body),
            })
            const data = (await res.json()) as TrialResponse

            if (!res.ok || ("ok" in data && data.ok === false)) {
                const msg =
                    ("ok" in data && (data.detail || data.error)) ||
                    "Something went wrong. Please try again."
                setError(String(msg))
                return
            }

            setResponse(data)
            // Cache the response so "Back to my results" from the upgrade
            // page can restore this view without re-running the engine.
            // Single-session, 1-hour ttl — cleared on handleStartOver.
            // resume_text is included so the cache-restore path can re-
            // populate `resume` state, which Start Over then carries
            // back to the landing page.
            try {
                sessionStorage.setItem(
                    RESULT_CACHE_KEY,
                    JSON.stringify({
                        response: data,
                        resume_text: r,
                        isOpenFlow: useOpenEndpoint,
                        email: e || null,
                        ts: Date.now(),
                    })
                )
            } catch {}

            // Fire scan_result_viewed pixel events. Fires only on real run
            // completions, not on cache restore (which doesn't reach here).
            // Verdict + score included so we can segment funnel by outcome.
            const sr =
                data && "result" in data && data.result ? data.result : null
            if (sr) {
                const verdict = String(sr.decision || "").toUpperCase()
                const score = typeof sr.score === "number" ? sr.score : null
                const utm = getEventUtmParams()
                const viewEventId = genEventId()
                try {
                    if (typeof window !== "undefined" && (window as any).fbq) {
                        ;(window as any).fbq(
                            "track",
                            "ViewContent",
                            {
                                content_name: "SIGNAL scan result",
                                content_category: "scan_result",
                                verdict,
                                score,
                                ...utm,
                            },
                            { eventID: viewEventId }
                        )
                    }
                } catch {}
                sendServerEvent({
                    event_name: "ViewContent",
                    event_id: viewEventId,
                    email: e || undefined,
                    custom_data: {
                        verdict,
                        score,
                        content_name: "SIGNAL scan result",
                        content_category: "scan_result",
                    },
                })
                try {
                    if (typeof window !== "undefined" && (window as any).gtag) {
                        ;(window as any).gtag("event", "scan_result_viewed", {
                            event_category: "funnel",
                            verdict,
                            score,
                            ...utm,
                        })
                    }
                } catch {}
            }
            try {
                window.scrollTo({ top: 0, behavior: "smooth" })
            } catch {}
        } catch (err: any) {
            setError(err?.message || "Network error. Please try again.")
        } finally {
            setLoading(false)
        }
    }

    // ── Mount: read email from URL, capture attribution, track view ─
    // Also: detect a v2-landing handoff payload and auto-run.
    React.useEffect(() => {
        if (typeof window === "undefined") return

        try {
            const params = new URLSearchParams(window.location.search)
            const e = params.get("email")
            if (e) setEmail(decodeURIComponent(e).trim().toLowerCase())
        } catch {}

        // Persistent session id for analytics
        const existing = localStorage.getItem("jobfit_session_id")
        const sessionId = existing || crypto.randomUUID()
        if (!existing) localStorage.setItem("jobfit_session_id", sessionId)

        // Capture attribution params (UTM + click IDs). Last-touch wins for
        // click IDs; first-touch wins for landing_page and referrer.
        captureAttribution()

        // Page view
        fetch(API_BASE + "/api/track", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                session_id: sessionId,
                page_path: window.location.pathname || "/jobfit-trial",
                page_name: "trial_jobfit_page",
                referrer: document.referrer || null,
                utm_source: ss("signal_utm_source") || null,
                utm_medium: ss("signal_utm_medium") || null,
                utm_campaign: ss("signal_utm_campaign") || null,
            }),
        }).catch(() => {})

        // ── v2 landing handoff: hydrate + auto-run open-endpoint flow ──
        // Checked BEFORE the result cache. A fresh handoff means the user
        // just submitted a NEW form on the landing page and wants a fresh
        // scan — restoring a cached result here would silently show the
        // prior scan instead of running the new one.
        // Same non-destructive read the loading initializer used, so the two
        // paths cannot diverge. This effect is the SOLE owner of removeItem
        // and runAnalysis.
        const handoff = peekFreshOpenFlowHandoff()
        if (handoff) {
            // Single-use: clear so back-button reloads don't re-trigger the
            // analysis.
            try {
                sessionStorage.removeItem(OPEN_FLOW_HANDOFF_KEY)
            } catch {}
            // A fresh submit supersedes any prior-scan cache.
            try {
                sessionStorage.removeItem(RESULT_CACHE_KEY)
            } catch {}
            // Mirror values into state so the input view (used as error
            // fallback) shows them pre-filled if the run fails.
            setResume(handoff.resume)
            setJd(handoff.jd)
            setIsOpenFlow(true)
            // Fire-and-forget; runAnalysis owns its own loading state.
            void runAnalysis({
                email: "",
                resume: handoff.resume,
                jd: handoff.jd,
                useOpenEndpoint: true,
            })
            return
        }

        // ── Cached result restore (back-nav from /signal/upgrade) ──
        // Only reached when there is NO fresh handoff above. This is the
        // "Back to my results" path — user returned from the upgrade page
        // and we restore their just-completed scan from cache to avoid a
        // re-run that would over-charge Haiku.
        try {
            const cachedRaw = sessionStorage.getItem(RESULT_CACHE_KEY)
            if (cachedRaw) {
                const cached = JSON.parse(cachedRaw) as {
                    response?: TrialResponse
                    resume_text?: string
                    isOpenFlow?: boolean
                    email?: string | null
                    ts?: number
                }
                const cts = typeof cached.ts === "number" ? cached.ts : 0
                const cFresh = cts && Date.now() - cts < RESULT_CACHE_MAX_AGE_MS
                if (cFresh && cached.response) {
                    setResponse(cached.response)
                    if (cached.resume_text) setResume(cached.resume_text)
                    if (cached.isOpenFlow) setIsOpenFlow(true)
                    if (cached.email) setEmail(cached.email)
                    return
                }
                // Stale: drop it so it doesn't linger past TTL.
                sessionStorage.removeItem(RESULT_CACHE_KEY)
            }
        } catch {}
    }, [])

    // ── Submit (manual input path) ──────────────────────────────────
    //
    // Stays on the open endpoint when the page was entered via the v2
    // landing handoff — open-flow retries from the fallback input view
    // would otherwise hit the gated endpoint and demand an email the user
    // never provided.
    async function handleAnalyze() {
        await runAnalysis({
            email: email.trim().toLowerCase(),
            resume: resume.trim(),
            jd: jd.trim(),
            useOpenEndpoint: isOpenFlow,
        })
    }

    // ── Upgrade ─────────────────────────────────────────────────────
    //
    // Open-flow users (came in via the v2 landing handoff, no email): every
    // CTA writes a handoff payload to sessionStorage and navigates to the
    // new /signal/upgrade page, which owns email capture and Stripe
    // redirect. Gated-flow users (entered with ?email= in the URL) keep
    // the existing direct-to-Stripe path unchanged.
    //
    // Funnel events:
    //   - Both paths fire AddToCart + upgrade_clicked here (upgrade
    //     intent). cta_source identifies which button was clicked.
    //   - Gated-flow ALSO fires InitiateCheckout + checkout_initiated
    //     here because their next step IS Stripe (they skip the upgrade
    //     page). Open-flow's InitiateCheckout fires on the upgrade page
    //     when they actually submit email.
    async function handleUpgrade(source: string) {
        if (upgradeLoading) return

        const sr = response?.result
        const verdict = sr ? String(sr.decision || "").toUpperCase() : null
        const score = sr && typeof sr.score === "number" ? sr.score : null
        const utm = getEventUtmParams()

        const addToCartEventId = genEventId()
        try {
            if (typeof window !== "undefined" && (window as any).fbq) {
                ;(window as any).fbq(
                    "track",
                    "AddToCart",
                    {
                        value: 99,
                        currency: "USD",
                        content_name: "SIGNAL 90-day access",
                        cta_source: source,
                        verdict,
                        score,
                        ...utm,
                    },
                    { eventID: addToCartEventId }
                )
            }
        } catch {}
        sendServerEvent({
            event_name: "AddToCart",
            event_id: addToCartEventId,
            email: email.trim().toLowerCase() || undefined,
            custom_data: {
                value: 99,
                currency: "USD",
                content_name: "SIGNAL 90-day access",
                cta_source: source,
                verdict: verdict || undefined,
                score,
            },
        })
        try {
            if (typeof window !== "undefined" && (window as any).gtag) {
                ;(window as any).gtag("event", "upgrade_clicked", {
                    event_category: "funnel",
                    cta_source: source,
                    verdict,
                    score,
                    ...utm,
                })
            }
        } catch {}

        if (!isOpenFlow) {
            // Gated path goes direct to Stripe — fire InitiateCheckout +
            // checkout_initiated here since there's no upgrade-page step.
            const initEventId = genEventId()
            try {
                if (typeof window !== "undefined" && (window as any).fbq) {
                    ;(window as any).fbq(
                        "track",
                        "InitiateCheckout",
                        {
                            value: 99,
                            currency: "USD",
                            content_name: "SIGNAL 90-day access",
                            verdict,
                            score,
                            ...utm,
                        },
                        { eventID: initEventId }
                    )
                }
            } catch {}
            sendServerEvent({
                event_name: "InitiateCheckout",
                event_id: initEventId,
                email: email.trim().toLowerCase() || undefined,
                custom_data: {
                    value: 99,
                    currency: "USD",
                    content_name: "SIGNAL 90-day access",
                    verdict: verdict || undefined,
                    score,
                },
            })
            try {
                if (typeof window !== "undefined" && (window as any).gtag) {
                    ;(window as any).gtag("event", "checkout_initiated", {
                        event_category: "funnel",
                        verdict,
                        score,
                        ...utm,
                    })
                }
            } catch {}
            try {
                if (typeof window !== "undefined" && (window as any).gtag) {
                    ;(window as any).gtag("event", "conversion", {
                        send_to: "AW-11125129027/AMw8CMHsraUcEMP-77gp",
                        value: 99.0,
                        currency: "USD",
                    })
                }
            } catch {}
        }

        if (isOpenFlow) {
            try {
                if (response) {
                    sessionStorage.setItem(
                        UPGRADE_HANDOFF_KEY,
                        JSON.stringify({
                            result: response.result,
                            locks: response.locks,
                            upgrade: response.upgrade,
                            cta_source: source,
                            ts: Date.now(),
                        })
                    )
                }
            } catch {}
            window.top!.location.replace(UPGRADE_PAGE_URL)
            return
        }

        // Gated path (email present, existing flow) ─────────────────────
        const e = email.trim().toLowerCase()
        if (!e) {
            window.top!.location.replace(LANDING_FALLBACK)
            return
        }
        setUpgradeLoading(true)
        try {
            const r = await fetch(API_BASE + "/api/checkout/create-session", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    email: e,
                    source,
                    ...getAttributionSnapshot(),
                }),
            })
            const j = await r.json()
            if (j.url) {
                window.open(j.url, "_blank")
            } else {
                window.top!.location.replace(LANDING_FALLBACK)
            }
        } catch {
            window.top!.location.replace(LANDING_FALLBACK)
        } finally {
            setUpgradeLoading(false)
        }
    }

    function handleStartOver() {
        // Stash the resume text so the landing page can pre-fill the
        // upload box on mount — Start Over should send the user back to
        // the landing form with their resume already attached so they
        // only need to paste a new JD.
        try {
            const cur = (resume || "").trim()
            if (cur.length >= MIN_RESUME_CHARS) {
                sessionStorage.setItem(
                    LANDING_RESUME_CARRYOVER_KEY,
                    JSON.stringify({ resume_text: cur, ts: Date.now() })
                )
            }
        } catch {}
        // Clear in-page caches so the trial page doesn't auto-restore
        // the old scan if the user navigates back later.
        try {
            sessionStorage.removeItem(RESULT_CACHE_KEY)
        } catch {}
        try {
            sessionStorage.removeItem(OPEN_FLOW_HANDOFF_KEY)
        } catch {}
        // Navigate to the v2 landing page. State clearing here is
        // unnecessary — the page unmounts on navigation.
        if (typeof window !== "undefined" && window.top) {
            window.top.location.replace(NEW_LANDING_URL)
        } else if (typeof window !== "undefined") {
            window.location.replace(NEW_LANDING_URL)
        }
    }

    // ── Render switch ───────────────────────────────────────────────
    return (
        <div style={page}>
            <PageShell />
            <style>{`
                @keyframes trialDotPulse {
                    0%, 100% { opacity: 1; transform: scale(1); }
                    50%      { opacity: 0.55; transform: scale(1.35); }
                }
                @keyframes trialSpin {
                    from { transform: rotate(0deg); }
                    to   { transform: rotate(360deg); }
                }
                @keyframes trialBounce {
                    0%, 100% { transform: translateY(0); }
                    50%      { transform: translateY(4px); }
                }
            `}</style>
            <Orbs hasResult={!!response} />
            <div style={{ position: "relative" as const, zIndex: 1 }}>
                <Topbar
                    response={response}
                    email={email}
                    onStartOver={handleStartOver}
                />

                {loading ? (
                    <>
                        <ThreeStageLoader />
                        <LoadingValueContent />
                    </>
                ) : !response ? (
                    <InputView
                        email={email}
                        resume={resume}
                        setResume={setResume}
                        jd={jd}
                        setJd={setJd}
                        error={error}
                        onAnalyze={handleAnalyze}
                        isOpenFlow={isOpenFlow}
                    />
                ) : response?.result?.perception ? (
                    <NewHandwritingResults
                        perception={response.result.perception}
                        onUpgrade={handleUpgrade}
                        upgradeLoading={upgradeLoading}
                        onStartOver={handleStartOver}
                    />
                ) : (
                    <ResultsView
                        response={response}
                        onUpgrade={handleUpgrade}
                        upgradeLoading={upgradeLoading}
                        onStartOver={handleStartOver}
                    />
                )}
            </div>
        </div>
    )
}

function ResultsView(props: {
    response: TrialResponse
    onUpgrade: (source: string) => void
    upgradeLoading: boolean
    onStartOver: () => void
}) {
    const { response, onUpgrade, upgradeLoading, onStartOver } = props

    if ("ok" in response && response.ok === false) return null // typed exhaustiveness

    if (response.status === "out_of_credits") {
        return (
            <OutOfCreditsView
                upgrade={response.upgrade}
                locks={response.locks}
                onUpgrade={onUpgrade}
                upgradeLoading={upgradeLoading}
            />
        )
    }

    // status === "ok" — fresh run OR cached repeat
    return (
        <SuccessView
            result={response.result}
            locked={response.locked}
            lockReason={response.lock_reason}
            locks={response.locks}
            upgrade={response.upgrade}
            onUpgrade={onUpgrade}
            upgradeLoading={upgradeLoading}
            onStartOver={onStartOver}
        />
    )
}

function HwPenUnderline(props: { color: string }) {
    return (
        <svg
            viewBox="0 0 200 7"
            preserveAspectRatio="none"
            style={{
                position: "absolute" as const,
                left: 0,
                right: 0,
                bottom: -3,
                width: "100%",
                height: 7,
                overflow: "visible",
                pointerEvents: "none" as const,
            }}
        >
            <path
                d="M2,4 C 28,1.5 54,5.5 80,3.5 S 132,2 158,4 S 188,5.5 198,3"
                fill="none"
                stroke={props.color}
                strokeWidth={2}
                strokeLinecap="round"
            />
        </svg>
    )
}
function HwPenCircle(props: { color: string }) {
    // Hand-drawn loop with near-vertical sides (not a tight ellipse): with
    // preserveAspectRatio="none" + height tracking the wrapped block, the sides
    // stay at the left/right edges across the FULL height, so a 2-3 line wrapped
    // marked_line is fully encircled — not just the first line.
    return (
        <svg
            viewBox="0 0 200 60"
            preserveAspectRatio="none"
            style={{
                position: "absolute" as const,
                left: -8,
                top: -6,
                width: "calc(100% + 16px)",
                height: "calc(100% + 12px)",
                overflow: "visible",
                pointerEvents: "none" as const,
            }}
        >
            <path
                d="M14,9 C 70,3 140,4 190,9 C 198,22 198,40 191,52 C 138,58 60,57 11,52 C 3,40 3,21 10,9 C 16,5 34,4 52,5"
                fill="none"
                stroke={props.color}
                strokeWidth={1.8}
                strokeLinecap="round"
                strokeLinejoin="round"
                vectorEffect="non-scaling-stroke"
            />
        </svg>
    )
}

    email: string
    onStartOver: () => void
}) {
    const { response, email, onStartOver } = props
    const hasResult =
        !!response && "status" in response && response.status === "ok"
    return (
        <div style={topbar}>
            <div>
                <div style={brand}>
                    SIGNAL<span style={{ color: C.orange }}>—</span>
                </div>
                <div style={brandSub}>BY WORKFORCE READY NOW</div>
            </div>
            <div
                style={{
                    fontSize: 11,
                    color: C.dim,
                    display: "flex",
                    alignItems: "center",
                    gap: 14,
                }}
            >
                {email ? <span style={{ color: C.muted }}>{email}</span> : null}
                {hasResult ? (
                    <span
                        style={{ color: C.blue, cursor: "pointer" }}
                        onClick={onStartOver}
                    >
                        Run another →
                    </span>
                ) : null}
            </div>
        </div>
    )
}

// ── ThreeStageLoader ────────────────────────────────────────────────────────
// Replaces LoadingBar. Three timed stages with checkmark advancement:
//   - Reading your resume (active 0–3s)
//   - Analyzing the job (active 3–7s)
//   - Generating your SIGNAL report (active 7s+, stays active until parent
//     unmounts us when the API response arrives)
// Self-managed timer; no marketing copy. After 20s elapsed, shows a subtle
// "still working…" hint. Failures are surfaced through the parent's error
// state, which falls back to the input view with the error message.

type StageStatus = "pending" | "active" | "complete"

function ThreeStageLoader() {
    const [elapsed, setElapsed] = React.useState(0)

    React.useEffect(() => {
        const startedAt = Date.now()
        const iv = setInterval(() => {
            setElapsed(Date.now() - startedAt)
        }, 200)
        return () => clearInterval(iv)
    }, [])

    function statusFor(idx: number): StageStatus {
        // [stage1End, stage2End, stage3End=Infinity]
        const ends = [3000, 7000, Number.POSITIVE_INFINITY]
        const starts = [0, 3000, 7000]
        if (elapsed >= ends[idx]) return "complete"
        if (elapsed >= starts[idx]) return "active"
        return "pending"
    }

    const showStillWorking = elapsed >= 20000

    return (
        <div style={loaderContainer}>
            <LoaderStage label="Reading your resume" status={statusFor(0)} />
            <LoaderStage label="Analyzing the job" status={statusFor(1)} />
            <LoaderStage
                label="Generating your SIGNAL report"
                status={statusFor(2)}
            />
            {showStillWorking ? (
                <div style={stillWorkingHint}>still working…</div>
            ) : null}
        </div>
    )
}

// sessionStorage read with try/catch — private mode + SSR safe.
function ss(key: string): string {
    try {
        return sessionStorage.getItem(key) || ""
    } catch {
        return ""
    }
}

function getCookie(name: string): string {
    if (typeof document === "undefined") return ""
    const m = document.cookie.match(
        new RegExp("(?:^|; )" + name.replace(/[-.]/g, "\\$&") + "=([^;]*)")
    )
    return m ? decodeURIComponent(m[1]) : ""
}

// First-touch landing/referrer; last-touch UTMs and click IDs.
function captureAttribution() {
    if (typeof window === "undefined") return
    try {
        const url = new URL(window.location.href)
        const lastTouch = (k: string) => url.searchParams.get(k)
        const writeIfPresent = (key: string, value: string | null) => {
            if (value) sessionStorage.setItem(key, value)
        }
        writeIfPresent("signal_utm_source", lastTouch("utm_source"))
        writeIfPresent("signal_utm_medium", lastTouch("utm_medium"))
        writeIfPresent("signal_utm_campaign", lastTouch("utm_campaign"))
        writeIfPresent("signal_utm_content", lastTouch("utm_content"))
        writeIfPresent("signal_utm_term", lastTouch("utm_term"))
        writeIfPresent("signal_fbclid", lastTouch("fbclid"))
        writeIfPresent("signal_ttclid", lastTouch("ttclid"))
        writeIfPresent("signal_gclid", lastTouch("gclid"))
        if (!sessionStorage.getItem("signal_landing_page")) {
            sessionStorage.setItem(
                "signal_landing_page",
                window.location.pathname || ""
            )
        }
        if (!sessionStorage.getItem("signal_referrer")) {
            sessionStorage.setItem("signal_referrer", document.referrer || "")
        }
    } catch {}
}

function getAttributionSnapshot() {
    return {
        utm_source: ss("signal_utm_source"),
        utm_medium: ss("signal_utm_medium"),
        utm_campaign: ss("signal_utm_campaign"),
        utm_content: ss("signal_utm_content"),
        utm_term: ss("signal_utm_term"),
        landing_page: ss("signal_landing_page"),
        referrer: ss("signal_referrer"),
        fbclid: ss("signal_fbclid"),
        ttclid: ss("signal_ttclid"),
        gclid: ss("signal_gclid"),
        fbp: getCookie("_fbp"),
        fbc: getCookie("_fbc"),
        ttp: getCookie("_ttp"),
    }
}

// Subset of attribution attached as custom params on client-side Meta /
// gtag events. UTMs only; click IDs and fbp/fbc/ttp ride the Stripe

function getEventUtmParams() {
    return {

// Server-side mirror of the browser pixel. Same event_id rides both
// fbq's eventID and the server CAPI call so Meta dedups the pair inside
// its 7-day (event_name, event_id) window. Fire-and-forget — browser
// pixel has fired before this network call resolves.
type FunnelEventName = "Lead" | "ViewContent" | "AddToCart" | "InitiateCheckout"
type FunnelCustomData = {
    value?: number
    currency?: string
    verdict?: string
    score?: number | null
    cta_source?: string
    content_name?: string
    content_category?: string
}

function genEventId(): string {
    try {
        if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
            return crypto.randomUUID()
        }
    } catch {}
    return (
        Date.now().toString(36) + "-" + Math.random().toString(36).slice(2, 10)
    )
}

function sendServerEvent(args: {
    event_name: FunnelEventName
    event_id: string
    email?: string
    custom_data?: FunnelCustomData
}) {
    try {
        const sessionId =
            (typeof localStorage !== "undefined" &&
                localStorage.getItem("jobfit_session_id")) ||
            ""
        const attribution = getAttributionSnapshot()
        const body = {
            event_name: args.event_name,
            event_id: args.event_id,
            event_time_sec: Math.floor(Date.now() / 1000),
            email: args.email || undefined,
            session_id: sessionId,
            ...attribution,
            custom_data: args.custom_data || {},
        }
        const url = API_BASE + "/api/conversions/event"
        // fetch with keepalive:true + credentials:"omit". sendBeacon would
        // be simpler but forces credentials:"include" which requires the
        // API to enable Access-Control-Allow-Credentials globally — not
        // wanted on a bearer-token-auth API.
        fetch(url, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
            keepalive: true,
            credentials: "omit",
        }).catch(() => {})
    } catch {}
}
